    """Demand generator — Poisson process synthetic customer arrivals."""

    from ccam_sim.models import SimState, DemandArrival


    def demand_generator(state: SimState) -> SimState:
        """Generate synthetic customer demand via Poisson process.

        For each demand block, sample interarrival times from exponential
        distribution with rate = block.rate (orders per hour).
        Converts interarrival time to integer microseconds.
        Continues until the block's end_hour exceeds total simulation time
        or cumulative arrivals exceed the block's budget.

        All randomness uses state.rng — never random module directly.
        """
        rng = state.rng
        scenario = state.scenario

        for block in scenario.demand_pattern:
            rate = float(block.rate)  # orders per hour
            if rate <= 0:
                continue

            # Convert block time window to microseconds from simulation start
            block_start_us = int(block.start_hour * 3_600_000_000)  # hours → microseconds
            block_end_us = int(block.end_hour * 3_600_000_000)
            sim_end_us = int(scenario.simulation.total_hours * 3_600_000_000)

            # Clamp block end to simulation end
            actual_end_us = min(block_end_us, sim_end_us)

            # Generate arrivals
            t = block_start_us
            while t < actual_end_us:
                # expovariate(lambd=1.0/rate): mean interarrival = 1/rate hours
                # Convert to microseconds: 3_600_000_000 microseconds per hour
                # lambd must be 1.0 / mean, where mean = 3600.0 / rate seconds
                # Actually expovariate(lambd) returns exponential with mean 1/lambd
                # If rate = 30 orders/hour, mean interarrival = 3600/30 = 120 seconds
                mean_seconds = 3600.0 / rate
                if mean_seconds <= 0:
                    break
                interarrival_seconds = rng.expovariate(1.0 / mean_seconds)
                interarrival_us = int(interarrival_seconds * 1_000_000)
                if interarrival_us < 1:
                    interarrival_us = 1  # minimum 1 microsecond

                t += interarrival_us
                if t < actual_end_us:
                    state.events.append(DemandArrival(time_us=t))

        return state
    ```

    """Order generator — converts synthetic demand into structured orders."""

    import random
    from ccam_sim.models import SimState, DemandArrival, OrderPlaced


    def order_generator(state: SimState) -> SimState:
        """Convert each DemandArrival event into an OrderPlaced event.

        For each DemandArrival in the event log:
        1. Assign a sequential order_id
        2. Select items based on the current demand block's item_weights
        3. Each order gets 1-5 items (randint), each item selected by weighted choice
        4. Create OrderPlaced event and add to current_orders

        Uses state.rng for all randomness.
        """
        rng = state.rng
        scenario = state.scenario

        # Find current demand block based on simulation time
        time_hours = state.time_us / 3_600_000_000.0

        # Build item_id → MenuItem lookup
        menu_map = {item.id: item for item in scenario.menu}

        # Process demand arrivals
        demand_arrivals = [e for e in state.events if isinstance(e, DemandArrival)]
        # Filter only arrivals that haven't been converted yet (no OrderPlaced for them)
        # Track the next order_id from existing order events
        existing_orders = [e for e in state.events if isinstance(e, OrderPlaced)]
        next_order_id = len(existing_orders) + 1

        for arrival in demand_arrivals:
            # Determine which demand block this arrival falls in
            arrival_hours = arrival.time_us / 3_600_000_000.0
            active_block = None
            for block in scenario.demand_pattern:
                if block.start_hour <= arrival_hours < block.end_hour:
                    active_block = block
                    break
            if active_block is None:
                # If no block matches, use the last block
                if scenario.demand_pattern:
                    active_block = scenario.demand_pattern[-1]
                else:
                    continue

            # Select 1-5 items per order
            num_items = rng.randint(1, 5)
            items = []
            if active_block.item_weights:
                # Weighted random choice with replacement
                item_ids = list(active_block.item_weights.keys())
                weights = list(active_block.item_weights.values())
                selected_ids = rng.choices(item_ids, weights=weights, k=num_items)
                for item_id in selected_ids:
                    menu_item = menu_map.get(item_id)
                    if menu_item:
                        items.append({
                            "item_id": item_id,
                            "name": menu_item.name,
                            "prep_time_seconds": menu_item.prep_time_seconds,
                            "station": menu_item.station,
                            "ingredients": dict(menu_item.ingredients),
                        })
            elif scenario.menu:
                # Fallback: uniform random from menu
                for _ in range(num_items):
                    item = rng.choice(scenario.menu)
                    items.append({
                        "item_id": item.id,
                        "name": item.name,
                        "prep_time_seconds": item.prep_time_seconds,
                        "station": item.station,
                        "ingredients": dict(item.ingredients),
                    })

            if items:
                order_event = OrderPlaced(
                    time_us=arrival.time_us,
                    order_id=next_order_id,
                    items=items,
                )
                state.events.append(order_event)
                state.current_orders.append({
                    "order_id": next_order_id,
                    "items": items,
                    "arrival_time_us": arrival.time_us,
                    "station_prep": {},  # populated by kitchen: station -> total prep time
                })
                next_order_id += 1

        return state
    ```

    """Kitchen engine — station load simulation, prep-time accumulation, bottleneck detection.

    Simulation uses fixed discrete ticks. Each tick:
    1. Calculate tick duration
    2. For each pending order, assign items to their stations
    3. Track prep time accumulation per station
    4. Detect bottlenecks (backlog > 3x tick capacity)
    5. Complete orders when all items have sufficient prep time accumulated
    """

    from ccam_sim.models import SimState, StationState, OrderCompleted, BottleneckEvent


    def kitchen_engine(state: SimState) -> SimState:
        """Process one tick of kitchen simulation."""
        scenario = state.scenario
        tick_duration_seconds = scenario.simulation.tick_interval_minutes * 60.0

        # Initialize station states if first run
        if not state.station_states:
            for station in scenario.stations:
                state.station_states[station.id] = StationState(
                    station_id=station.id,
                    staff_count=station.staff_count,
                )

        for order in state.current_orders:
            if not order.get("station_prep"):
                order["station_prep"] = {}

            for item in order["items"]:
                station_id = item.get("station", "")
                prep_seconds = item.get("prep_time_seconds", 0)

                if station_id not in order["station_prep"]:
                    order["station_prep"][station_id] = 0.0
                order["station_prep"][station_id] += prep_seconds

                if "prep_progress" not in item:
                    item["prep_progress"] = 0.0

                station_state = state.station_states.get(station_id)
                if station_state:
                    capacity_this_tick = station_state.staff_count * tick_duration_seconds
                    remaining = prep_seconds - item["prep_progress"]
                    station_state.current_backlog_seconds += remaining
                    processed = min(station_state.current_backlog_seconds, capacity_this_tick)
                    station_state.current_backlog_seconds -= processed
                    station_state.total_utilization_seconds += processed

                    if remaining > 0:
                        fraction = processed / max(remaining, 0.001)
                        item["prep_progress"] += remaining * fraction

                    if station_state.current_backlog_seconds > capacity_this_tick * 3:
                        state.events.append(BottleneckEvent(
                            time_us=state.time_us,
                            station_id=station_id,
                            backlog_seconds=station_state.current_backlog_seconds,
                            capacity_seconds=capacity_this_tick,
                            duration_seconds=tick_duration_seconds,
                        ))
                        station_state.bottleneck_seconds += tick_duration_seconds

        # Check for completed orders
        completed_orders = []
        for i, order in enumerate(state.current_orders):
            all_completed = True
            total_prep = 0.0
            station_ids = set()
            for item in order["items"]:
                prep = item.get("prep_time_seconds", 0)
                progress = item.get("prep_progress", 0)
                if progress < prep:
                    all_completed = False
                    break
                total_prep += prep
                station_ids.add(item.get("station", ""))

            if all_completed:
                completion_time_minutes = (state.time_us - order["arrival_time_us"]) / 60_000_000.0
                state.events.append(OrderCompleted(
                    time_us=state.time_us,
                    order_id=order["order_id"],
                    completion_time_minutes=completion_time_minutes,
                    total_prep_time_seconds=total_prep,
                    station_ids=list(station_ids),
                ))
                completed_orders.append(i)

        for i in reversed(completed_orders):
            state.current_orders.pop(i)

        return state
    ```

    """Stock engine — ingredient consumption, stockout detection, waste tracking."""

    from ccam_sim.models import (
        SimState, OrderCompleted, StockConsumed, StockoutEvent, WasteEvent, OrderPlaced,
    )


    def stock_engine(state: SimState) -> SimState:
        """Process stock consumption for all completed orders in this tick."""
        processed_order_ids = set()
        for event in state.events:
            if isinstance(event, StockConsumed):
                processed_order_ids.add(event.order_id)

        scenario = state.scenario
        menu_map = {item.id: item for item in scenario.menu}

        if not state.stock_levels:
            state.stock_levels = dict(scenario.stock)

        completed_events = [
            e for e in state.events
            if isinstance(e, OrderCompleted) and e.order_id not in processed_order_ids
        ]

        for completed in completed_events:
            order_placed = None
            for e in state.events:
                if isinstance(e, OrderPlaced) and e.order_id == completed.order_id:
                    order_placed = e
                    break
            if order_placed is None:
                continue

            for item in order_placed.items:
                item_id = item.get("item_id", "")
                menu_item = menu_map.get(item_id)
                if menu_item is None:
                    continue

                for ingredient, qty_needed in menu_item.ingredients.items():
                    current_stock = state.stock_levels.get(ingredient, 0.0)
                    if current_stock >= qty_needed:
                        state.stock_levels[ingredient] = current_stock - qty_needed
                        state.events.append(StockConsumed(
                            time_us=state.time_us,
                            ingredient=ingredient, qty=qty_needed,
                            order_id=completed.order_id, unit_cost=0.0,
                        ))
                    elif current_stock > 0:
                        state.stock_levels[ingredient] = 0.0
                        state.events.append(StockConsumed(
                            time_us=state.time_us,
                            ingredient=ingredient, qty=current_stock,
                            order_id=completed.order_id, unit_cost=0.0,
                        ))
                        state.events.append(StockoutEvent(
                            time_us=state.time_us, ingredient=ingredient,
                            requested_qty=qty_needed, available_qty=current_stock,
                            order_id=completed.order_id,
                        ))
                        state.events.append(WasteEvent(
                            time_us=state.time_us, ingredient=ingredient,
                            qty=qty_needed - current_stock, waste_cost=0.0,
                            reason="stockout_shortfall",
                        ))
                    else:
                        state.events.append(StockoutEvent(
                            time_us=state.time_us, ingredient=ingredient,
                            requested_qty=qty_needed, available_qty=0.0,
                            order_id=completed.order_id,
                        ))
                        state.events.append(WasteEvent(
                            time_us=state.time_us, ingredient=ingredient,
                            qty=qty_needed, waste_cost=0.0,
                            reason="stockout_unavailable",
                        ))

        return state
    ```

    """KPI engine — computes simulation metrics from event log into typed KPIResult."""

    from ccam_sim.models import SimState, KPIResult
    from ccam_sim.models.events import (
        StockoutEvent, WasteEvent, BottleneckEvent, OrderCompleted, StockConsumed,
        OrderPlaced,
    )


    def kpi_engine(state: SimState) -> SimState:
        """Compute KPIs and event count aggregates from the simulation event log.

        Returns state with state.kpis set to a typed KPIResult instance (not a dict).

        Rate-based KPIs:
        - stockout_frequency: stockout events per simulation hour
        - waste_rate: total waste qty / total consumed qty
        - bottleneck_frequency: total bottleneck minutes per simulation hour
        - order_completion_time_mean_minutes: mean time from arrival to completion
        - throughput_orders_per_hour: completed orders per simulation hour

        Event count aggregates (computed from event log):
        - total_orders_placed, total_orders_completed, total_stockouts,
          total_waste_events, total_bottleneck_minutes, total_revenue
        """
        events = state.events
        total_hours = state.scenario.simulation.total_hours
        if total_hours <= 0:
            total_hours = 1.0  # avoid division by zero

        # Count events by type
        stockout_events = [e for e in events if isinstance(e, StockoutEvent)]
        waste_events = [e for e in events if isinstance(e, WasteEvent)]
        bottleneck_events = [e for e in events if isinstance(e, BottleneckEvent)]
        completed_orders = [e for e in events if isinstance(e, OrderCompleted)]
        placed_orders = [e for e in events if isinstance(e, OrderPlaced)]

        # Total consumed qty
        consumed_events = [e for e in events if isinstance(e, StockConsumed)]
        total_consumed = sum(e.qty for e in consumed_events)
        total_waste_qty = sum(e.qty for e in waste_events)

        # Bottleneck minutes total
        total_bottleneck_minutes = sum(
            e.duration_seconds / 60.0 for e in bottleneck_events
        )

        # Mean order completion time
        completion_times = [e.completion_time_minutes for e in completed_orders]
        mean_completion = (
            sum(completion_times) / len(completion_times)
            if completion_times else 0.0
        )

        # Build typed KPIResult instance (not a dict)
        state.kpis = KPIResult(
            # Rate-based KPIs
            stockout_frequency=len(stockout_events) / total_hours,
            waste_rate=total_waste_qty / max(total_consumed, 1.0),
            bottleneck_frequency=total_bottleneck_minutes / total_hours,
            order_completion_time_minutes=mean_completion,
            throughput_orders_per_hour=len(completed_orders) / total_hours,
            # Event count aggregates
            total_orders_placed=len(placed_orders),
            total_orders_completed=len(completed_orders),
            total_stockouts=len(stockout_events),
            total_waste_events=len(waste_events),
            total_bottleneck_minutes=total_bottleneck_minutes,
            total_revenue=0.0,  # Phase 1: no pricing model; Phase 5+ adds revenue tracking
        )

        return state
    ```

    This is the main entry point for running simulations. Imported by:
    - CLI (ccam_sim.cli) for the `simulate` command
    - Phase 3 FastAPI for REST API simulation
    """

    from ccam_sim.models import Scenario, SimState
    from ccam_sim.rand import create_rng
    from ccam_sim.engines.demand import demand_generator
    from ccam_sim.engines.order import order_generator
    from ccam_sim.engines.kitchen import kitchen_engine
    from ccam_sim.engines.stock import stock_engine
    from ccam_sim.engines.kpi import kpi_engine
    from ccam_sim.safety import check_per_tick, SafetyError


    def run_simulation(scenario: Scenario, seed: int) -> SimState:
        """Run deterministic simulation from scenario YAML data.

        Args:
            scenario: Parsed and validated Scenario dataclass.
            seed: Integer RNG seed for deterministic replay.

        Returns:
            SimState with populated kpis and events after full simulation.

        Raises:
            SafetyError: If a per-tick safety check fails during simulation.
        """
        # Initialize state with seeded RNG and metadata fields
        rng = create_rng(seed)
        state = SimState(
            rng=rng,
            scenario=scenario,
            seed=seed,
        )

        total_hours = scenario.simulation.total_hours
        tick_interval_minutes = scenario.simulation.tick_interval_minutes
        max_steps = scenario.simulation.max_steps
        tick_duration_us = tick_interval_minutes * 60 * 1_000_000
        total_duration_us = int(total_hours * 3_600_000_000)

        # Phase 1: Demand generation (event-based, runs once)
        state = demand_generator(state)

        # Phase 2: Order generation (event-based, runs once)
        state = order_generator(state)

        # Phases 3-5: Tick-based simulation loop
        state.step = 0
        state.time_us = 0

        while state.time_us < total_duration_us and state.step < max_steps:
            state.step += 1

            # Kitchen engine: process orders through stations
            state = kitchen_engine(state)

            # Stock engine: consume ingredients for completed orders
            state = stock_engine(state)

            # Gate 2: Per-tick safety enforcement
            check_per_tick(state)

            # Advance time by one tick
            state.time_us += tick_duration_us

        # Phase 6: KPI computation (runs once at end)
        state = kpi_engine(state)

        # Populate metadata fields consumed by output formatters
        state.simulated_hours = total_hours

        return state
    ```
