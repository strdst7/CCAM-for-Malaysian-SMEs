    def resolve_scenario_path(path: str) -> str:
        """Resolve scenario path, checking common extensions.

        If path ends with .yaml or .yml, return as-is.
        Otherwise try path + '.yaml', then path + '.yml'.
        Raises FileNotFoundError if none exist.
        """
        p = Path(path)
        if p.suffix in (".yaml", ".yml"):
            return path
        for ext in (".yaml", ".yml"):
            candidate = p.with_suffix(ext)
            if candidate.exists():
                return str(candidate)
        raise FileNotFoundError(
            f"Scenario file not found: {path} (tried {path}.yaml, {path}.yml)"
        )
    ```

    """CCAM Simulator CLI — Click-based command-line interface."""

    import json
    import sys
    from pathlib import Path

    import click

    from ccam_sim.models import load_scenario, ScenarioLoadError
    from ccam_sim.models.loader import resolve_scenario_path
    from ccam_sim.safety import validate_scenario, SafetyError


    @click.group()
    def cli():
        """CCAM Restaurant Simulator — deterministic digital twin CLI.

        Run scenario YAML files through the full simulation pipeline and
        receive structured KPI output plus a human-readable summary.
        """
        pass


    @cli.command()
    @click.argument("scenario", type=click.Path(exists=True, dir_okay=False))
    @click.option("--seed", type=int, default=42, show_default=True,
                  help="RNG seed for deterministic replay")
    @click.option("--output", "-o", type=click.Path(dir_okay=False),
                  help="Write KPI JSON to file instead of stdout")
    @click.option("--quiet", "-q", is_flag=True,
                  help="Suppress human-readable summary to stderr")
    def simulate(scenario, seed, output, quiet):
        """Run deterministic simulation from SCENARIO YAML.

        SCENARIO is a path to a .yaml or .yml scenario file.
        """
        try:
            # Resolve path with extension fallback
            resolved = resolve_scenario_path(scenario)

            # Load and parse scenario YAML
            scenario_obj = load_scenario(resolved)

            # Safety Gate 1: static validation at load time
            validate_scenario(scenario_obj)

            # Run simulation (pipeline imported here — contract fulfilled in Plan 03)
            from ccam_sim.pipeline import run_simulation
            state = run_simulation(scenario_obj, seed=seed)

            # Format outputs
            from ccam_sim.output import format_kpi_json, format_summary

            # Always output JSON to stdout
            kpi_json = format_kpi_json(state)
            if output:
                output_path = Path(output)
                output_path.write_text(kpi_json)
            click.echo(kpi_json)

            # Human-readable summary to stderr (doesn't interfere with JSON piping)
            if not quiet:
                summary = format_summary(state)
                click.echo(summary, err=True)

        except FileNotFoundError as e:
            click.echo(f"Error: {e}", err=True)
            sys.exit(1)
        except ScenarioLoadError as e:
            click.echo(f"Error: Invalid scenario — {e}", err=True)
            sys.exit(1)
        except SafetyError as e:
            click.echo(f"Safety Error: {e}", err=True)
            sys.exit(1)
        except Exception as e:
            click.echo(f"Unexpected Error: {e}", err=True)
            sys.exit(1)
    ```

    ```python
    """Safety layer — static validation (Gate 1) and per-tick enforcement (Gate 2).

    Gate 1 is called once after scenario loading, before simulation begins.
    Gate 2 is called every tick during simulation to enforce runtime invariants.
    """

    from typing import List, Optional
    from ccam_sim.models import Scenario, SimState


    class SafetyError(Exception):
        """Raised when a safety rule is violated."""


    # ── Gate 1: Static scenario validation ──────────────────────────────

    def validate_scenario(scenario: Scenario) -> None:
        """Gate 1 — Validate scenario before simulation begins.

        Checks:
        1. No negative initial stock levels
        2. No station with staff_count < 1 (>120% utilization means no capacity)
        3. max_steps > 0 (infinite loop guard)

        Raises:
            SafetyError: On first violation with descriptive message.
        """
        _check_negative_stock(scenario)
        _check_station_utilization(scenario)
        _check_max_steps(scenario)


    def _check_negative_stock(scenario: Scenario) -> None:
        """Rule: No ingredient can have negative initial stock."""
        for ingredient, qty in scenario.stock.items():
            if qty < 0:
                raise SafetyError(
                    f"Negative stock for '{ingredient}': {qty}. "
                    f"All stock levels must be >= 0."
                )


    def _check_station_utilization(scenario: Scenario) -> None:
        """Rule: No station can have staff_count less than 1.

        staff_count=0 means the station has no capacity, making utilization
        infinite. staff_count must be >= 1 per the spec's "no overload >120%"
        requirement — you can't overload a station that doesn't exist.
        """
        for station in scenario.stations:
            if station.staff_count < 1:
                raise SafetyError(
                    f"Station '{station.id}' has staff_count={station.staff_count}. "
                    f"Each station must have at least 1 staff member."
                )


    def _check_max_steps(scenario: Scenario) -> None:
        """Rule: max_steps must be > 0 (infinite loop guard)."""
        if scenario.simulation.max_steps < 1:
            raise SafetyError(
                f"simulation.max_steps={scenario.simulation.max_steps}. "
                f"max_steps must be greater than 0 to prevent infinite loops."
            )


    # ── Gate 2: Per-tick enforcement ────────────────────────────────────

    def check_per_tick(state: SimState) -> None:
        """Gate 2 — Enforce runtime invariants after each simulation tick.

        Checks:
        1. No ingredient stock went negative this tick
        2. Step limit not exceeded (infinite loop guard)

        Station backlog bounds check (_check_backlog_bounds) is defined but
        not called here — it will be enabled in Plan 03 when station_states
        are populated by the kitchen engine.

        Raises:
            SafetyError: On first violation with descriptive message.
        """
        _check_stock_non_negative(state)
        _check_step_limit(state)


    def _check_stock_non_negative(state: SimState) -> None:
        """Rule: No ingredient stock level can go below 0."""
        for ingredient, qty in state.stock_levels.items():
            if qty < 0:
                raise SafetyError(
                    f"Stock for '{ingredient}' went negative: {qty}. "
                    f"Stock level must never drop below 0."
                )


    def _check_step_limit(state: SimState) -> None:
        """Rule: step count must not exceed max_steps (infinite loop guard)."""
        if state.step > state.scenario.simulation.max_steps:
            raise SafetyError(
                f"Step limit exceeded: {state.step} > {state.scenario.simulation.max_steps}. "
                f"Simulation terminated — possible infinite loop."
            )


    def _check_backlog_bounds(state: SimState) -> None:
        """Rule: No station backlog can exceed 10x its per-tick capacity.

        Defined but not called in Plan 02. Will be called in Plan 03 after
        the kitchen engine populates station_states.
        """
        tick_duration_seconds = state.scenario.simulation.tick_interval_minutes * 60.0
        for station_id, st in state.station_states.items():
            capacity_per_tick = st.staff_count * tick_duration_seconds
            if capacity_per_tick > 0 and st.current_backlog_seconds > capacity_per_tick * 10:
                raise SafetyError(
                    f"Station '{station_id}' backlog ({st.current_backlog_seconds:.1f}s) "
                    f"exceeded 10x capacity ({capacity_per_tick * 10:.1f}s). "
                    f"Unstable backlog detected."
                )
    ```

    """Output formatters: JSON and human-readable summary."""

    import json
    from typing import Any

    from ccam_sim.models import SimState


    def format_kpi_json(state: SimState) -> str:
        """Format simulation results as structured JSON.

        Produces:
        {
          "simulation": {
            "scenario_file": str or null,
            "seed": int,
            "total_steps": int,
            "simulated_hours": float,
            "completed_orders": int,
            "total_revenue": float
          },
          "kpis": {
            "stockout_frequency": float,
            "waste_rate": float,
            "bottleneck_frequency": float,
            "order_completion_time_minutes": float,
            "throughput_orders_per_hour": float
          },
          "events_summary": {
            "total_orders_placed": int,
            "total_orders_completed": int,
            "total_stockouts": int,
            "total_waste_events": int,
            "total_bottleneck_minutes": float,
            "total_revenue": float
          }
        }
        """
        kpi = state.kpis
        result: dict[str, Any] = {
            "simulation": {
                "scenario_file": getattr(state, "scenario_file", None),
                "seed": state.seed,
                "total_steps": state.step,
                "simulated_hours": state.simulated_hours,
                "completed_orders": kpi.total_orders_completed if kpi else 0,
                "total_revenue": round(kpi.total_revenue, 2) if kpi else 0.0,
            },
            "kpis": {
                "stockout_frequency": round(kpi.stockout_frequency, 4) if kpi else 0.0,
                "waste_rate": round(kpi.waste_rate, 4) if kpi else 0.0,
                "bottleneck_frequency": round(kpi.bottleneck_frequency, 4) if kpi else 0.0,
                "order_completion_time_minutes": round(kpi.order_completion_time_minutes, 2) if kpi else 0.0,
                "throughput_orders_per_hour": round(kpi.throughput_orders_per_hour, 2) if kpi else 0.0,
            },
            "events_summary": {
                "total_orders_placed": kpi.total_orders_placed if kpi else 0,
                "total_orders_completed": kpi.total_orders_completed if kpi else 0,
                "total_stockouts": kpi.total_stockouts if kpi else 0,
                "total_waste_events": kpi.total_waste_events if kpi else 0,
                "total_bottleneck_minutes": round(kpi.total_bottleneck_minutes, 2) if kpi else 0.0,
                "total_revenue": round(kpi.total_revenue, 2) if kpi else 0.0,
            },
        }
        return json.dumps(result, indent=2)

            ("Orders Completed", str(kpi.total_orders_completed if kpi else 0)),
        ]
        for label, value in sim_data:
            padded = f"│  {label:26s}{value:>26s} │"

            lines.append(padded)
        lines.append("├─────────────────────────────────────────────────────────────┤")
        lines.append("│  EVENTS                                                     │")
        lines.append("├─────────────────────────────────────────────────────────────┤")
        evt_data = [

            ("Orders Completed", str(kpi.total_orders_completed if kpi else 0)),
