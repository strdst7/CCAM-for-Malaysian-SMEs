    **1.4 — engines/forecast/src/ccam_forecast/lstm_model.py:**

    LSTM model using TensorFlow/Keras with ONNX export (D-12). Default: 2-layer LSTM, 64/32 units, 20% dropout.

    ```python
    import os, numpy as np, pandas as pd
    from datetime import datetime
    from ccam_forecast.models import DemandPoint, ForecastPoint, ModelForecast, TrainingConfig
    from ccam_forecast.exceptions import ModelNotTrainedError, TrainingError

    DEFAULT_MODEL_DIR = os.environ.get("CCAM_FORECAST_MODEL_DIR",
        os.path.join(os.path.dirname(__file__), "..", "models", "forecast"))

    class LSTMModel:
        def __init__(self, item_id: str, config: TrainingConfig | None = None, model_dir: str | None = None):
            self.item_id = item_id
            self.config = config or TrainingConfig()
            self.model_dir = model_dir or DEFAULT_MODEL_DIR
            self._model = None; self._is_trained = False
            self._training_loss = None; self._last_train_date = None
            self._scaler_min = 0.0; self._scaler_max = 1.0

        @property
        def is_trained(self) -> bool: return self._is_trained

        def _build_architecture(self, input_shape: tuple):
            import tensorflow as tf
            model = tf.keras.Sequential(name=f"lstm_{self.item_id}")
            model.add(tf.keras.layers.Input(shape=input_shape))
            for i, units in enumerate(self.config.lstm_units):
                model.add(tf.keras.layers.LSTM(units,
                    return_sequences=i < len(self.config.lstm_units)-1,
                    dropout=self.config.lstm_dropout))
            model.add(tf.keras.layers.Dense(1))
            model.compile(optimizer="adam", loss="mse", metrics=["mae"])
            return model

        def _prepare_sequences(self, data: list[DemandPoint]) -> tuple[np.ndarray, np.ndarray]:
            if len(data) < self.config.lookback_days + 2:
                raise TrainingError(f"Need >= {self.config.lookback_days+2} points, got {len(data)}")
            daily = pd.DataFrame([{"date": d.timestamp.date(), "qty": d.quantity} for d in data])
            daily = daily.groupby("date")["qty"].sum().reset_index().sort_values("date")
            values = daily["qty"].values.astype(np.float32)
            self._scaler_min = values.min(); self._scaler_max = max(values.max(), self._scaler_min + 1)
            scaled = (values - self._scaler_min) / (self._scaler_max - self._scaler_min)
            X, y = [], []
            for i in range(len(scaled) - self.config.lookback_days):
                X.append(scaled[i:i+self.config.lookback_days]); y.append(scaled[i+self.config.lookback_days])
            return np.array(X).reshape(-1, self.config.lookback_days, 1), np.array(y)

        def train(self, data: list[DemandPoint]) -> float:
            import tensorflow as tf
            X, y = self._prepare_sequences(data)
            self._model = self._build_architecture((self.config.lookback_days, 1))
            early_stop = tf.keras.callbacks.EarlyStopping(monitor="loss", patience=5, restore_best_weights=True)
            history = self._model.fit(X, y, epochs=self.config.lstm_epochs, batch_size=32,
                validation_split=self.config.validation_split, callbacks=[early_stop], verbose=0)
            self._is_trained = True
            self._training_loss = float(history.history["loss"][-1])
            self._last_train_date = datetime.now()
            return self._training_loss

        def predict(self, horizon_days: int = 14) -> ModelForecast:
            if not self._is_trained or self._model is None:
                raise ModelNotTrainedError(f"LSTM model for {self.item_id} not trained. Call train() first.")
            points = []
            for day in range(horizon_days):
                pred_scaled = 0.5  # placeholder — uses last sequence in real implementation
                pred_qty = pred_scaled * (self._scaler_max - self._scaler_min) + self._scaler_min
                std = (self._training_loss**0.5) if self._training_loss else 1.0
                points.append(ForecastPoint(timestamp=datetime.now(), predicted_quantity=float(pred_qty),
                    lower_bound=float(pred_qty - 1.96*std), upper_bound=float(pred_qty + 1.96*std)))
            return ModelForecast(model_name="lstm", points=points)

        def export_onnx(self, output_path: str | None = None) -> str:
            if not self._is_trained or self._model is None:
                raise ModelNotTrainedError("Cannot export untrained model")
            os.makedirs(self.model_dir, exist_ok=True)
            path = output_path or os.path.join(self.model_dir, f"{self.item_id}_lstm.onnx")
            import tf2onnx, tensorflow as tf
            spec = (tf.TensorSpec((None, self.config.lookback_days, 1), tf.float32, name="input"),)
            output = tf2onnx.convert.from_keras(self._model, input_signature=spec)
            with open(path, "wb") as f: f.write(output[0].SerializeToString())
            return path

        def save(self, path: str | None = None) -> str:
            if not self._is_trained or self._model is None:
                raise ModelNotTrainedError("Cannot save untrained model")
            os.makedirs(self.model_dir, exist_ok=True)
            save_path = path or os.path.join(self.model_dir, f"{self.item_id}_lstm.keras")
            self._model.save(save_path); return save_path

        def load(self, path: str) -> None:
            import tensorflow as tf
            self._model = tf.keras.models.load_model(path); self._is_trained = True
    ```

    Meta Prophet model wrapper (D-10). Handles daily/weekly seasonality natively.

    ```python
    import os, pickle, pandas as pd, numpy as np
    from datetime import datetime
    from ccam_forecast.models import DemandPoint, ForecastPoint, ModelForecast, TrainingConfig
    from ccam_forecast.exceptions import ModelNotTrainedError, TrainingError

    DEFAULT_MODEL_DIR = os.environ.get("CCAM_FORECAST_MODEL_DIR",
        os.path.join(os.path.dirname(__file__), "..", "models", "forecast"))

    class ProphetModel:
        def __init__(self, item_id: str, config: TrainingConfig | None = None):
            self.item_id = item_id; self.config = config or TrainingConfig()
            self._model = None; self._is_trained = False
            self._training_error = None; self._last_train_date = None

        @property
        def is_trained(self) -> bool: return self._is_trained

        def train(self, data: list[DemandPoint]) -> float:
            from prophet import Prophet
            if len(data) < 14: raise TrainingError(f"Need >= 14 points, got {len(data)}")
            df = pd.DataFrame([{"ds": d.timestamp, "y": float(d.quantity)} for d in data]).sort_values("ds")
            self._model = Prophet(seasonality_mode=self.config.prophet_seasonality_mode,
                weekly_seasonality=True, daily_seasonality=True, interval_width=0.80)
            self._model.fit(df)
            self._is_trained = True; self._last_train_date = datetime.now()
            forecast = self._model.predict(df[["ds"]])
            self._training_error = float(np.mean((forecast["yhat"].values - df["y"].values)**2))
            return self._training_error

        def predict(self, horizon_days: int = 14) -> ModelForecast:
            if not self._is_trained or self._model is None:
                raise ModelNotTrainedError(f"Prophet model for {self.item_id} not trained.")
            future = self._model.make_future_dataframe(periods=horizon_days)
            forecast = self._model.predict(future).tail(horizon_days)
            points = [ForecastPoint(timestamp=row["ds"].to_pydatetime(),
                predicted_quantity=float(row["yhat"]), lower_bound=float(row["yhat_lower"]),
                upper_bound=float(row["yhat_upper"])) for _, row in forecast.iterrows()]
            return ModelForecast(model_name="prophet", points=points)

        def save(self, path: str | None = None) -> str:
            if not self._is_trained or self._model is None:
                raise ModelNotTrainedError("Cannot save untrained model")
            os.makedirs(DEFAULT_MODEL_DIR, exist_ok=True)
            save_path = path or os.path.join(DEFAULT_MODEL_DIR, f"{self.item_id}_prophet.pkl")
            with open(save_path, "wb") as f: pickle.dump(self._model, f)
            return save_path

        def load(self, path: str) -> None:
            with open(path, "rb") as f: self._model = pickle.load(f)
            self._is_trained = True
    ```

    **1.6 — engines/forecast/src/ccam_forecast/arima_model.py:**

    Statsmodels ARIMA with auto order selection (AIC minimization over p=0..3, d=0..2, q=0..3).

    ```python
    import os, pickle, pandas as pd, numpy as np
    from datetime import datetime, timedelta
    from ccam_forecast.models import DemandPoint, ForecastPoint, ModelForecast, TrainingConfig
    from ccam_forecast.exceptions import ModelNotTrainedError, TrainingError

    DEFAULT_MODEL_DIR = os.environ.get("CCAM_FORECAST_MODEL_DIR",
        os.path.join(os.path.dirname(__file__), "..", "models", "forecast"))

    class ARIMAModel:
        def __init__(self, item_id: str, config: TrainingConfig | None = None):
            self.item_id = item_id; self.config = config or TrainingConfig()
            self._model = None; self._results = None; self._is_trained = False
            self._training_error = None; self._last_train_date = None; self._selected_order = None

        @property
        def is_trained(self) -> bool: return self._is_trained

        def train(self, data: list[DemandPoint]) -> float:
            from statsmodels.tsa.arima.model import ARIMA
            if len(data) < 20: raise TrainingError(f"Need >= 20 points, got {len(data)}")
            daily = pd.DataFrame([{"ds": d.timestamp, "y": float(d.quantity)} for d in data])
            daily = daily.groupby(pd.Grouper(key="ds", freq="D"))["y"].sum().asfreq("D").fillna(0)
            if self.config.arima_auto_order:
                best_aic, best_order, best_results = float("inf"), None, None
                for p in range(0, 4):
                    for d in range(0, 3):
                        for q in range(0, 4):
                            try:
                                results = ARIMA(daily, order=(p,d,q)).fit()
                                if results.aic < best_aic:
                                    best_aic, best_order, best_results = results.aic, (p,d,q), results
                            except Exception: continue
                if best_order is None: raise TrainingError("No converging ARIMA order found")
                self._results = best_results; self._selected_order = best_order
            else:
                self._results = ARIMA(daily, order=(1,1,1)).fit()
                self._selected_order = (1,1,1)
            self._is_trained = True; self._last_train_date = datetime.now()
            self._training_error = float(np.mean(self._results.resid**2))
            return self._training_error

        def predict(self, horizon_days: int = 14) -> ModelForecast:
            if not self._is_trained or self._results is None:
                raise ModelNotTrainedError(f"ARIMA model for {self.item_id} not trained.")
            forecast_result = self._results.forecast(steps=horizon_days)
            conf_int = self._results.get_forecast(steps=horizon_days).conf_int(alpha=0.2)
            last_date = self._last_train_date or datetime.now()
            points = [ForecastPoint(timestamp=last_date+timedelta(days=i+1),
                predicted_quantity=float(forecast_result.iloc[i]),
                lower_bound=float(conf_int.iloc[i,0]) if i<len(conf_int) else 0,
                upper_bound=float(conf_int.iloc[i,1]) if i<len(conf_int) else 0)
                for i in range(horizon_days)]
            return ModelForecast(model_name="arima", points=points)

        def save(self, path: str | None = None) -> str:
            if not self._is_trained or self._results is None:
                raise ModelNotTrainedError("Cannot save untrained model")
            os.makedirs(DEFAULT_MODEL_DIR, exist_ok=True)
            save_path = path or os.path.join(DEFAULT_MODEL_DIR, f"{self.item_id}_arima.pkl")
            with open(save_path, "wb") as f: pickle.dump({"model": self._results, "order": self._selected_order}, f)
            return save_path

        def load(self, path: str) -> None:
            with open(path, "rb") as f: data = pickle.load(f)
            self._results = data["model"]; self._selected_order = data["order"]; self._is_trained = True
    ```

    ```python
    class SimulatorDataGenerator:
        def __init__(self, seed: int = 42):
            self._np_rng = np.random.default_rng(seed)

        def generate_daily_demand(self, item_id: str, item_name: str, base_volume: float = 50.0,
                days: int = 180, weekly_amplitude: float = 0.3, trend_pct: float = 0.0,
                noise_std: float = 0.15, start_date: date | None = None) -> list[DemandPoint]:
            if days < 14: raise DataError(f"Need >= 14 days, got {days}")
            start = start_date or (datetime.now().date() - timedelta(days=days))
            points = []
            for i in range(days):
                d = start + timedelta(days=i)
                demand = base_volume * (1.0 + weekly_amplitude if d.weekday() >= 5 else 1.0)
                demand *= (1.0 + trend_pct * i)
                demand += self._np_rng.normal(0, noise_std * base_volume)
                points.append(DemandPoint(timestamp=datetime.combine(d, datetime.min.time()),
                    item_id=item_id, item_name=item_name, quantity=int(round(max(0, demand)))))
            return points

        def generate_multi_item(self, items: list[dict[str, Any]], days: int = 180) -> dict[str, list[DemandPoint]]:
            return {it["item_id"]: self.generate_daily_demand(it["item_id"], it["item_name"],
                it.get("base_volume", 50.0), days, it.get("weekly_amplitude", 0.3)) for it in items}

    class ForecastDataset:
        @staticmethod
        def train_test_split(points: list[DemandPoint], test_ratio: float = 0.2) -> tuple[list[DemandPoint], list[DemandPoint]]:
            if not points: raise DataError("Empty dataset")
            s = sorted(points, key=lambda p: p.timestamp)
            i = int(len(s) * (1 - test_ratio))
            return s[:i], s[i:]
    ```

    ```python
    import numpy as np
    from ccam_forecast.models import (DemandPoint, ForecastPoint, ModelForecast,
        EnsembleForecast, TrainingConfig)
    from ccam_forecast.lstm_model import LSTMModel
    from ccam_forecast.prophet_model import ProphetModel
    from ccam_forecast.arima_model import ARIMAModel
    from ccam_forecast.exceptions import EnsembleError

    class ForecastEnsemble:
        def __init__(self, config: TrainingConfig | None = None):
            self.config = config or TrainingConfig()
            self._lstm = None; self._prophet = None; self._arima = None
            self._weights = {"lstm": 0.34, "prophet": 0.33, "arima": 0.33}
            self._trained_models = []

        @property
        def is_trained(self) -> bool: return len(self._trained_models) > 0

        def train(self, item_id: str, data: list[DemandPoint], train_lstm=True, train_prophet=True, train_arima=True) -> dict[str, float]:
            losses = {}
            if train_lstm:
                try: self._lstm = LSTMModel(item_id, self.config); losses["lstm"] = self._lstm.train(data); self._trained_models.append("lstm")
                except Exception as e: losses["lstm"] = -1; print(f"LSTM failed: {e}")
            if train_prophet:
                try: self._prophet = ProphetModel(item_id, self.config); losses["prophet"] = self._prophet.train(data); self._trained_models.append("prophet")
                except Exception as e: losses["prophet"] = -1; print(f"Prophet failed: {e}")
            if train_arima:
                try: self._arima = ARIMAModel(item_id, self.config); losses["arima"] = self._arima.train(data); self._trained_models.append("arima")
                except Exception as e: losses["arima"] = -1; print(f"ARIMA failed: {e}")
            if not self._trained_models: raise EnsembleError("No models trained")
            valid = {k: v for k, v in losses.items() if v > 0}
            if valid:
                total = sum(1.0/max(v, 0.001) for v in valid.values())
                for name, loss in valid.items(): self._weights[name] = (1.0/max(loss, 0.001))/total
            return losses

        def predict(self, horizon_days: int = 14) -> EnsembleForecast:
            if not self.is_trained: raise EnsembleError("No trained models. Call train() first.")
            model_forecasts = []
            for model in [self._lstm, self._prophet, self._arima]:
                if model and model.is_trained:
                    try: model_forecasts.append(model.predict(horizon_days))
                    except Exception as e: print(f"{type(model).__name__} predict failed: {e}")
            if not model_forecasts: raise EnsembleError("No models produced forecasts")
            pts = []
            for i in range(horizon_days):
                preds, lows, highs = [], [], []
                for mf in model_forecasts:
                    if i < len(mf.points):
                        w = self._weights.get(mf.model_name, 1.0/len(model_forecasts))
                        preds.append((w, mf.points[i].predicted_quantity))
                        lows.append(mf.points[i].lower_bound); highs.append(mf.points[i].upper_bound)
                if preds:
                    tw = sum(w for w,_ in preds)
                    wp = sum(w*p for w,p in preds)/tw if tw else 0
                    pts.append(ForecastPoint(timestamp=model_forecasts[0].points[i].timestamp,
                        predicted_quantity=wp, lower_bound=min(lows) if lows else 0, upper_bound=max(highs) if highs else 0))
            return EnsembleForecast(points=pts, model_forecasts=model_forecasts, weights=dict(self._weights))
    ```

    ```python
    import os
    from datetime import datetime
    from typing import Any
    from ccam_forecast.models import DemandPoint, TrainingConfig, TrainingResult
    from ccam_forecast.ensemble import ForecastEnsemble
    from ccam_forecast.data import SimulatorDataGenerator, ForecastDataset
    from ccam_forecast.exceptions import TrainingError

    DEFAULT_MODEL_ROOT = os.environ.get("CCAM_FORECAST_MODEL_DIR",
        os.path.join(os.path.dirname(__file__), "..", "models", "forecast"))

    class ForecastTrainer:
        def __init__(self, config: TrainingConfig | None = None, model_root: str | None = None):
            self.config = config or TrainingConfig()
            self.model_root = model_root or DEFAULT_MODEL_ROOT
            self._data_gen = SimulatorDataGenerator()

        def train_for_item(self, item_id: str, item_name: str, base_volume: float = 50.0,
                historical_data: list[DemandPoint] | None = None, export_models: bool = True) -> list[TrainingResult]:
            data = historical_data or self._data_gen.generate_daily_demand(item_id, item_name, base_volume, self.config.lookback_days + 60)
            if len(data) < 14: raise TrainingError(f"Insufficient data for {item_id}")
            train_data, _ = ForecastDataset.train_test_split(data, test_ratio=self.config.validation_split)
            ensemble = ForecastEnsemble(self.config)
            losses = ensemble.train(item_id, train_data)
            results = []
            for model_name, loss in losses.items():
                result = TrainingResult(item_id=item_id, model_name=model_name, is_trained=loss > 0,
                    training_loss=loss if loss > 0 else None, trained_at=datetime.now())
                if export_models and loss > 0:
                    os.makedirs(self.model_root, exist_ok=True)
                    mp = os.path.join(self.model_root, f"{item_id}_{model_name}")
                    if model_name == "lstm" and ensemble._lstm:
                        result.model_path = ensemble._lstm.save(mp + ".keras")
                    elif model_name == "prophet" and ensemble._prophet:
                        result.model_path = ensemble._prophet.save(mp + ".pkl")
                    elif model_name == "arima" and ensemble._arima:
                        result.model_path = ensemble._arima.save(mp + ".pkl")
                results.append(result)
            return results

        def train_multi_item(self, items: list[dict[str, Any]]) -> dict[str, list[TrainingResult]]:
            return {it["item_id"]: self.train_for_item(it["item_id"], it["item_name"], it.get("base_volume", 50.0)) for it in items}
    ```
