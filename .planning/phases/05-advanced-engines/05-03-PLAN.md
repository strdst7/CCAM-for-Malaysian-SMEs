    ```python
    """Pluggable supplier adapter interface (D-14).

    All supplier-specific implementations subclass SupplierAdapter and
    implement the four core methods.
    """
    from abc import ABC, abstractmethod
    from ccam_procurement.models import PurchaseOrder, POStatus, SupplierContact

    class SupplierAdapter(ABC):
        """Abstract base for supplier adapters."""

        @abstractmethod
        def order_ingredients(self, po: PurchaseOrder) -> bool:
            """Submit a purchase order to the supplier.

            Returns True if the order was accepted by the supplier system.
            """
            ...

        @abstractmethod
        def generate_po(self, po: PurchaseOrder) -> bytes:
            """Generate a purchase order document (PDF).

            Returns the document as raw bytes.
            """
            ...

        @abstractmethod
        def send_po(self, po: PurchaseOrder, document: bytes) -> bool:
            """Send the purchase order document to the supplier.

            Returns True if delivery was successful.
            """
            ...

        @abstractmethod
        def track_status(self, po_id: str) -> POStatus:
            """Check the current status of a purchase order with this supplier.

            Returns the current POStatus.
            """
            ...

        @property
        @abstractmethod
        def adapter_name(self) -> str:
            """Human-readable name for this adapter (e.g., 'email', 'sysfood')."""
            ...
    ```

    ```python
    """Purchase order PDF generator using reportlab.

    Generates a standardized PO document with:
    - PO header (ID, date, supplier info)
    - Line items table (item, qty, unit, unit price, total)
    - Totals section
    - Terms and notes
    """
    from io import BytesIO
    from datetime import datetime
    from reportlab.lib.pagesizes import A4
    from reportlab.lib.units import mm
    from reportlab.lib import colors
    from reportlab.platypus import (SimpleDocTemplate, Table, TableStyle,
        Paragraph, Spacer, HRFlowable)
    from reportlab.lib.styles import getSampleStyleSheet, ParagraphStyle
    from reportlab.lib.enums import TA_RIGHT, TA_CENTER
    from ccam_procurement.models import PurchaseOrder, POLineItem, POStatus
    from ccam_procurement.exceptions import POGenerationError

    class POGenerator:
        """Generates PDF purchase order documents."""

        def generate(self, po: PurchaseOrder) -> bytes:
            """Generate a PDF purchase order document.

            Args:
                po: The purchase order data.

            Returns:
                PDF document as bytes.
            """
            try:
                buf = BytesIO()
                doc = SimpleDocTemplate(buf, pagesize=A4,
                    leftMargin=20*mm, rightMargin=20*mm,
                    topMargin=20*mm, bottomMargin=20*mm)
                styles = getSampleStyleSheet()
                elements = []

                # Title
                elements.append(Paragraph(f"<b>PURCHASE ORDER</b>", styles["Title"]))
                elements.append(Spacer(1, 5*mm))

                # PO Info
                info_style = ParagraphStyle("Info", parent=styles["Normal"], fontSize=10)
                elements.append(Paragraph(f"PO #: {po.po_id}", info_style))
                elements.append(Paragraph(f"Date: {po.created_at.strftime('%Y-%m-%d %H:%M')}", info_style))
                elements.append(Paragraph(f"Status: {po.status.value}", info_style))
                elements.append(Paragraph(f"Supplier: {po.supplier_name}", info_style))
                elements.append(Spacer(1, 5*mm))
                elements.append(HRFlowable(width="100%", thickness=1))
                elements.append(Spacer(1, 3*mm))

                # Line Items Table
                table_data = [
                    ["Item", "Qty", "Unit", "Unit Price", "Total"],
                ]
                for item in po.line_items:
                    table_data.append([
                        item.ingredient_name,
                        str(item.quantity),
                        item.unit,
                        f"RM{item.unit_price_cents/100:.2f}",
                        f"RM{item.total_cents/100:.2f}",
                    ])

                # Totals row
                table_data.append(["", "", "", "TOTAL:", f"RM{po.total_cents/100:.2f}"])

                col_widths = [80*mm, 20*mm, 20*mm, 30*mm, 30*mm]
                table = Table(table_data, colWidths=col_widths)
                table.setStyle(TableStyle([
                    ("BACKGROUND", (0, 0), (-1, 0), colors.HexColor("#2c3e50")),
                    ("TEXTCOLOR", (0, 0), (-1, 0), colors.whitesmoke),
                    ("FONTSIZE", (0, 0), (-1, 0), 10),
                    ("FONTSIZE", (0, 1), (-1, -1), 9),
                    ("ALIGN", (1, 0), (-1, -1), "CENTER"),
                    ("ALIGN", (-1, 0), (-1, -1), "RIGHT"),
                    ("GRID", (0, 0), (-1, -2), 0.5, colors.grey),
                    ("LINEBELOW", (0, -1), (-1, -1), 1, colors.black),
                    ("BACKGROUND", (0, -1), (-2, -1), colors.HexColor("#ecf0f1")),
                ]))
                elements.append(table)
                elements.append(Spacer(1, 5*mm))

                # Notes
                if po.notes:
                    elements.append(Paragraph(f"<b>Notes:</b>", styles["Normal"]))
                    elements.append(Paragraph(po.notes, styles["Normal"]))

                # Build PDF
                doc.build(elements)
                pdf_bytes = buf.getvalue()
                buf.close()
                return pdf_bytes

            except Exception as e:
                raise POGenerationError(f"PDF generation failed: {e}")

    class SimplePOGenerator:
        """Simplified PO generator for lightweight output (text-based)."""

        def generate(self, po: PurchaseOrder) -> str:
            """Generate a plain-text purchase order summary."""
            lines = []
            lines.append("=" * 50)
            lines.append(f"PURCHASE ORDER #{po.po_id}")
            lines.append("=" * 50)
            lines.append(f"Date: {po.created_at.strftime('%Y-%m-%d %H:%M')}")
            lines.append(f"Supplier: {po.supplier_name}")
            lines.append(f"Status: {po.status.value}")
            lines.append("-" * 50)
            lines.append(f"{'Item':<30} {'Qty':<8} {'Price':<10}")
            lines.append("-" * 50)
            for item in po.line_items:
                lines.append(f"{item.ingredient_name:<30} {item.quantity:<8} RM{item.total_cents/100:.2f}")
            lines.append("-" * 50)
            lines.append(f"{'TOTAL':<38} RM{po.total_cents/100:.2f}")
            lines.append("=" * 50)
            return "\n".join(lines)
    ```

    ```python
    """Email-based supplier adapter for purchase order delivery (D-15).

    Sends PDF purchase orders via email using SendGrid as primary
    channel and SMTP as fallback.

    Supplier contact info is expected to be configured per-tenant and
    passed via SupplierContact object.
    """
    import os
    import smtplib
    from email.mime.multipart import MIMEMultipart
    from email.mime.base import MIMEBase
    from email.mime.text import MIMEText
    from email.utils import formatdate
    from email import encoders

    from ccam_procurement.supplier_abc import SupplierAdapter
    from ccam_procurement.models import PurchaseOrder, POStatus, SupplierContact
    from ccam_procurement.po_generator import POGenerator
    from ccam_procurement.exceptions import AdapterError


    class EmailSupplierAdapter(SupplierAdapter):
        """Email-based purchase order delivery.

        Uses SendGrid by default. Falls back to SMTP if SENDGRID_API_KEY is not set.
        """

        def __init__(
            self,
            supplier_contact: SupplierContact,
            from_email: str | None = None,
            sendgrid_api_key: str | None = None,
        ):
            self._contact = supplier_contact
            self._from_email = from_email or os.environ.get(
                "SENDGRID_FROM_EMAIL", "purchasing@ccam.local"
            )
            self._sendgrid_key = sendgrid_api_key or os.environ.get("SENDGRID_API_KEY", "")
            self._po_gen = POGenerator()

        @property
        def adapter_name(self) -> str:
            return "email"

        def order_ingredients(self, po: PurchaseOrder) -> bool:
            """Submit order via email (sends PO document)."""
            document = self.generate_po(po)
            return self.send_po(po, document)

        def generate_po(self, po: PurchaseOrder) -> bytes:
            """Generate PDF purchase order."""
            return self._po_gen.generate(po)

        def send_po(self, po: PurchaseOrder, document: bytes) -> bool:
            """Send PO document via email."""
            if self._sendgrid_key:
                return self._send_via_sendgrid(po, document)
            else:
                return self._send_via_smtp(po, document)

        def track_status(self, po_id: str) -> POStatus:
            """Email delivery doesn't support status tracking — returns sent status."""
            return POStatus.SENT

        def _send_via_sendgrid(self, po: PurchaseOrder, document: bytes) -> bool:
            """Send using SendGrid API."""
            try:
                from sendgrid import SendGridAPIClient
                from sendgrid.helpers.mail import (
                    Mail, Attachment, FileContent, FileName,
                    FileType, Disposition, ContentId, Email,
                )

                message = Mail(
                    from_email=Email(self._from_email),
                    to_emails=Email(self._contact.email),
                    subject=f"Purchase Order #{po.po_id} from CCAM",
                    plain_text_content=(
                        f"Dear {self._contact.name},\n\n"
                        f"Please find attached Purchase Order #{po.po_id} "
                        f"for a total of RM{po.total_cents/100:.2f}.\n\n"
                        f"Best regards,\nCCAM Procurement"
                    ),
                )

                # Attach PDF
                encoded = __import__("base64").b64encode(document).decode()
                attachment = Attachment(
                    FileContent(encoded),
                    FileName(f"PO_{po.po_id}.pdf"),
                    FileType("application/pdf"),
                    Disposition("attachment"),
                )
                message.add_attachment(attachment)

                sg = SendGridAPIClient(self._sendgrid_key)
                response = sg.send(message)
                return response.status_code in (200, 201, 202)

            except Exception as e:
                raise AdapterError(f"SendGrid delivery failed: {e}")

        def _send_via_smtp(self, po: PurchaseOrder, document: bytes) -> bool:
            """Send using direct SMTP (fallback)."""
            try:
                msg = MIMEMultipart()
                msg["From"] = self._from_email
                msg["To"] = self._contact.email
                msg["Subject"] = f"Purchase Order #{po.po_id} from CCAM"
                msg["Date"] = formatdate(localtime=True)

                body = MIMEText(
                    f"Dear {self._contact.name},\n\n"
                    f"Please find attached Purchase Order #{po.po_id} "
                    f"for a total of RM{po.total_cents/100:.2f}.\n\n"
                    f"Best regards,\nCCAM Procurement"
                )
                msg.attach(body)

                # Attach PDF
                part = MIMEBase("application", "pdf")
                part.set_payload(document)
                encoders.encode_base64(part)
                part.add_header(
                    "Content-Disposition",
                    f"attachment; filename=PO_{po.po_id}.pdf",
                )
                msg.attach(part)

                smtp_host = os.environ.get("SMTP_HOST", "localhost")
                smtp_port = int(os.environ.get("SMTP_PORT", "25"))
                smtp_user = os.environ.get("SMTP_USER", "")
                smtp_pass = os.environ.get("SMTP_PASS", "")

                with smtplib.SMTP(smtp_host, smtp_port) as server:
                    if smtp_user:
                        server.login(smtp_user, smtp_pass)
                    server.send_message(msg)

                return True

            except Exception as e:
                raise AdapterError(f"SMTP delivery failed: {e}")
    ```

    ```python
    """High-level procurement engine.

    Orchestrates:
    1. Create PO from request
    2. Generate PDF document
    3. Deliver via supplier adapter
    4. Return PO with status
    """
    from datetime import datetime
    from ccam_procurement.models import (
        PurchaseOrder, POLineItem, POStatus, POGenerateRequest, SupplierContact,
    )
    from ccam_procurement.supplier_abc import SupplierAdapter
    from ccam_procurement.email_adapter import EmailSupplierAdapter
    from ccam_procurement.exceptions import ProcurementError, SupplierNotFoundError

    class ProcurementEngine:
        """Procurement engine with pluggable supplier adapters."""

        def __init__(self):
            self._adapters: dict[str, SupplierAdapter] = {}

        def register_adapter(self, supplier_id: str, adapter: SupplierAdapter) -> None:
            """Register a supplier adapter for a specific supplier."""
            self._adapters[supplier_id] = adapter

        def create_purchase_order(
            self,
            request: POGenerateRequest,
            supplier_contact: SupplierContact,
            po_id: str | None = None,
        ) -> PurchaseOrder:
            """Create a purchase order and send it to the supplier.

            If no adapter is registered for the supplier, creates an email adapter
            using the provided supplier contact info.

            Args:
                request: PO generation request.
                supplier_contact: Supplier contact details.
                po_id: Optional PO ID (auto-generated if not provided).

            Returns:
                The created PurchaseOrder with updated status.
            """
            import uuid
            total_cents = sum(item.total_cents for item in request.line_items)

            po = PurchaseOrder(
                po_id=po_id or f"PO-{uuid.uuid4().hex[:12].upper()}",
                tenant_id=request.tenant_id,
                supplier_id=request.supplier_id,
                supplier_name=supplier_contact.name,
                line_items=request.line_items,
                total_cents=total_cents,
                status=POStatus.PENDING_APPROVAL,
                notes=request.notes,
            )

            # Get or create adapter
            adapter = self._adapters.get(request.supplier_id)
            if adapter is None:
                adapter = EmailSupplierAdapter(supplier_contact)

            # Generate and send
            try:
                document = adapter.generate_po(po)
                success = adapter.send_po(po, document)

                if success:
                    po.status = POStatus.SENT
                    po.sent_at = datetime.now()
                else:
                    po.status = POStatus.DRAFT

            except Exception as e:
                po.status = POStatus.DRAFT
                raise ProcurementError(f"PO delivery failed: {e}")

            return po

    def create_and_send_po(
        request: POGenerateRequest,
        supplier_contact: SupplierContact,
    ) -> PurchaseOrder:
        """Convenience function: create and send a purchase order."""
        engine = ProcurementEngine()
        return engine.create_purchase_order(request, supplier_contact)
    ```

    ```python
    """Email notification channel — SendGrid primary, SMTP fallback."""
    import os
    import smtplib
    import string
    from email.mime.text import MIMEText
    from email.utils import formatdate

    from ccam_notification.models import (Notification, NotificationTemplate,
        NotificationStatus, SendResult, NotificationChannel)
    from ccam_notification.exceptions import DeliveryError


    class EmailChannel:
        """Email notification delivery channel."""

        def __init__(
            self,
            from_email: str | None = None,
            sendgrid_api_key: str | None = None,
        ):
            self._from_email = from_email or os.environ.get(
                "SENDGRID_FROM_EMAIL", "noreply@ccam.local"
            )
            self._sendgrid_key = sendgrid_api_key or os.environ.get("SENDGRID_API_KEY", "")

        def send(self, notification: Notification, template: NotificationTemplate) -> SendResult:
            """Send a notification via email.

            Uses SendGrid if API key is configured, otherwise falls back to SMTP.
            """
            try:
                # Render template
                tmpl = string.Template(template.subject_template)
                subject = tmpl.safe_substitute(**notification.template_data)
                tmpl = string.Template(template.body_template)
                body = tmpl.safe_substitute(**notification.template_data)

                if self._sendgrid_key:
                    return self._send_via_sendgrid(notification, subject, body)
                else:
                    return self._send_via_smtp(notification, subject, body)

            except Exception as e:
                return SendResult(
                    success=False, channel=NotificationChannel.EMAIL,
                    notification_id=notification.id, error=str(e),
                )

        def _send_via_sendgrid(self, notification: Notification, subject: str, body: str) -> SendResult:
            from sendgrid import SendGridAPIClient
            from sendgrid.helpers.mail import Mail, Email
            message = Mail(
                from_email=Email(self._from_email),
                to_emails=Email(notification.user_id),  # user_id = email in v1
                subject=subject,
                plain_text_content=body,
            )
            sg = SendGridAPIClient(self._sendgrid_key)
            response = sg.send(message)
            return SendResult(
                success=response.status_code in (200, 201, 202),
                channel=NotificationChannel.EMAIL,
                notification_id=notification.id,
            )

        def _send_via_smtp(self, notification: Notification, subject: str, body: str) -> SendResult:
            msg = MIMEText(body)
            msg["From"] = self._from_email
            msg["To"] = notification.user_id
            msg["Subject"] = subject
            msg["Date"] = formatdate(localtime=True)

            smtp_host = os.environ.get("SMTP_HOST", "localhost")
            smtp_port = int(os.environ.get("SMTP_PORT", "25"))
            with smtplib.SMTP(smtp_host, smtp_port) as server:
                user = os.environ.get("SMTP_USER", "")
                if user:
                    server.login(user, os.environ.get("SMTP_PASS", ""))
                server.send_message(msg)

            return SendResult(
                success=True, channel=NotificationChannel.EMAIL,
                notification_id=notification.id,
            )
    ```

    ```python
    """In-app notification channel.

    Stores notifications that can be retrieved via API (SSE or polling).
    This module provides the in-memory store; backend persistence is
    added in Plan 04 via database models.
    """
    from datetime import datetime
    from typing import Any

    from ccam_notification.models import (Notification, NotificationTemplate,
        NotificationStatus, SendResult, NotificationChannel)


    class InAppChannel:
        """In-app notification store.

        Notifications are stored in memory and retrieved via the
        notification API. The backend (Plan 04) adds database persistence
        and SSE/polling endpoints.
        """

        def __init__(self):
            self._notifications: dict[str, Notification] = {}

        def send(self, notification: Notification, template: NotificationTemplate) -> SendResult:
            """Store an in-app notification."""
            import string
            import uuid

            # Generate body from template
            tmpl = string.Template(template.body_template if template.body_template else template.subject_template)
            rendered_body = tmpl.safe_substitute(**notification.template_data)

            notif = Notification(
                id=notification.id or f"notif-{uuid.uuid4().hex[:12]}",
                tenant_id=notification.tenant_id,
                user_id=notification.user_id,
                channel=NotificationChannel.INAPP,
                template_key=notification.template_key,
                template_data=notification.template_data,
                title=template.subject_template,
                body=rendered_body,
                priority=notification.priority,
                status=NotificationStatus.SENT,
                created_at=datetime.now(),
                sent_at=datetime.now(),
            )

            self._notifications[notif.id] = notif

            return SendResult(
                success=True,
                channel=NotificationChannel.INAPP,
                notification_id=notif.id,
            )

        def get_notifications(self, user_id: str, limit: int = 50) -> list[Notification]:
            """Get in-app notifications for a user, newest first."""
            user_notifs = [
                n for n in self._notifications.values()
                if n.user_id == user_id
            ]
            user_notifs.sort(key=lambda n: n.created_at, reverse=True)
            return user_notifs[:limit]

        def mark_read(self, notification_id: str) -> bool:
            """Mark an in-app notification as read."""
            if notification_id in self._notifications:
                self._notifications[notification_id].status = NotificationStatus.READ
                return True
            return False

        def count_unread(self, user_id: str) -> int:
            """Count unread notifications for a user."""
            return sum(
                1 for n in self._notifications.values()
                if n.user_id == user_id and n.status == NotificationStatus.SENT
            )
    ```

    Multi-channel notification router with template system.

    ```python
    """Notification router that selects and dispatches to channels.

    Routes notifications to the appropriate channel(s) based on
    the notification configuration and user preferences.
    """
    import string
    from typing import Any

    from ccam_notification.models import (Notification, NotificationTemplate,
        NotificationChannel, SendResult)
    from ccam_notification.email_channel import EmailChannel
    from ccam_notification.inapp_channel import InAppChannel
    from ccam_notification.exceptions import TemplateError, ChannelError


    class NotificationRouter:
        """Routes notifications to configured channels with template rendering."""

        def __init__(
            self,
            email_channel: EmailChannel | None = None,
            inapp_channel: InAppChannel | None = None,
        ):
            self._email = email_channel or EmailChannel()
            self._inapp = inapp_channel or InAppChannel()
            self._templates: dict[str, NotificationTemplate] = {}
            self._default_templates: dict[NotificationChannel, str] = {}

        def register_template(self, template: NotificationTemplate) -> None:
            """Register a notification template."""
            self._templates[template.key] = template

        def set_default_template(self, channel: NotificationChannel, template_key: str) -> None:
            """Set the default template key for a channel."""
            self._default_templates[channel] = template_key

        def send(
            self,
            notification: Notification,
            channels: list[NotificationChannel] | None = None,
        ) -> list[SendResult]:
            """Send a notification through one or more channels.

            Args:
                notification: The notification to send.
                channels: Channels to use. Defaults to the notification's own channel.

            Returns:
                List of SendResult (one per channel).
            """
            channels = channels or [notification.channel]
            results = []

            for channel in channels:
                # Get template
                template_key = notification.template_key or self._default_templates.get(channel, "")
                template = self._templates.get(template_key)

                if template is None and template_key:
                    raise TemplateError(f"Template '{template_key}' not found for channel {channel.value}")

                # Send via appropriate channel
                if channel == NotificationChannel.EMAIL:
                    result = self._email.send(notification, template or self._default_email_template())
                elif channel == NotificationChannel.INAPP:
                    result = self._inapp.send(notification, template or self._default_inapp_template())
                else:
                    raise ChannelError(f"Unsupported channel: {channel}")

                results.append(result)

            return results

        def _default_email_template(self) -> NotificationTemplate:
            return NotificationTemplate(
                key="_default_email",
                subject_template="${title}",
                body_template="${body}",
                channel=NotificationChannel.EMAIL,
            )

        def _default_inapp_template(self) -> NotificationTemplate:
            return NotificationTemplate(
                key="_default_inapp",
                subject_template="${title}",
                body_template="${body}",
                channel=NotificationChannel.INAPP,
            )
    ```

    ```python
    import uuid
    from datetime import datetime

    class NotificationEngine:
        """Notification engine with template management and multi-channel dispatch."""

        def __init__(self):
            self._router = NotificationRouter()

        @property
        def router(self) -> NotificationRouter:
            return self._router

        def notify(
            self,
            tenant_id: str,
            user_id: str,
            title: str,
            body: str,
            channel: NotificationChannel = NotificationChannel.INAPP,
            priority: NotificationPriority = NotificationPriority.NORMAL,
            template_key: str = "",
            template_data: dict[str, Any] | None = None,
        ) -> list[SendResult]:
            """Create and send a notification.

            Args:
                tenant_id: Tenant scope.
                user_id: Recipient identifier (email for email channel, user ID for in-app).
                title: Notification title.
                body: Notification body text.
                channel: Target channel.
                priority: Notification priority.
                template_key: Optional registered template key.
                template_data: Data for template rendering.

            Returns:
                List of SendResult.
            """
            notification = Notification(
                id=f"n-{uuid.uuid4().hex[:12]}",
                tenant_id=tenant_id,
                user_id=user_id,
                channel=channel,
                template_key=template_key,
                template_data=template_data or {"title": title, "body": body},
                title=title,
                body=body,
                priority=priority,
                status=NotificationStatus.PENDING,
                created_at=datetime.now(),
            )

            return self._router.send(notification, [channel])

        def notify_email(
            self,
            tenant_id: str,
            email: str,
            subject: str,
            body: str,
            template_data: dict[str, Any] | None = None,
        ) -> list[SendResult]:
            """Convenience: send an email notification."""
            return self.notify(
                tenant_id=tenant_id,
                user_id=email,
                title=subject,
                body=body,
                channel=NotificationChannel.EMAIL,
                template_data=template_data or {"title": subject, "body": body},
            )

            self,
            tenant_id: str,
            user_id: str,
            title: str,
            body: str,

            return self.notify(
                tenant_id=tenant_id,
                user_id=user_id,
                title=title,
                body=body,
                channel=NotificationChannel.INAPP,
            )

        title: str,

        engine = NotificationEngine()
        return engine.notify(tenant_id, user_id, title, body, channel)
    ```
