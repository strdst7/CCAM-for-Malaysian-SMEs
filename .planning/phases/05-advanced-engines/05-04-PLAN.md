    Parser webhook endpoint. Per D-03 wraps sync engine in async via run_in_executor.

    ```python
    import asyncio
    from fastapi import APIRouter, Depends, HTTPException, Request
    from pydantic import BaseModel, Field
    from ccam_backend.auth.middleware import verify_hmac
    from ccam_parser import parse_message
    from ccam_parser.models import ParsedOrder, ParsedItem

    router = APIRouter(tags=["parser"])

    class ParseRequest(BaseModel):
        message: str = Field(..., min_length=1, max_length=4096, description="Raw WhatsApp message text")

    class ParsedItemResponse(BaseModel):
        name: str
        quantity: int
        unit_price_cents: int | None = None
        confidence: float

    class ParseResponse(BaseModel):
        items: list[ParsedItemResponse]
        customer_name: str
        pickup_time: str
        special_instructions: str
        total_price_cents: int | None = None
        overall_confidence: float
        parser_source: str

    def _to_response(order: ParsedOrder) -> ParseResponse:
        return ParseResponse(
            items=[ParsedItemResponse(name=i.name, quantity=i.quantity,
                unit_price_cents=i.unit_price_cents, confidence=i.confidence) for i in order.items],
            customer_name=order.customer_name,
            pickup_time=order.pickup_time,
            special_instructions=order.special_instructions,
            total_price_cents=order.total_price_cents,
            overall_confidence=order.overall_confidence,
            parser_source=order.parser_source,
        )

    @router.post("/parse", response_model=ParseResponse)
    async def parse_whatsapp_message(request: Request, body: ParseRequest, auth: tuple = Depends(verify_hmac)):
        loop = asyncio.get_event_loop()
        order = await loop.run_in_executor(None, parse_message, body.message)
        return _to_response(order)
    ```

    **2.2 — backend/src/ccam_backend/api/menu.py:**

    Menu optimization endpoint. Wraps sync MenuEngine via run_in_executor. Per D-09, endpoint docs instruct callers to validate via sim before deployment.

    ```python
    import asyncio
    from fastapi import APIRouter, Depends, HTTPException, Request
    from pydantic import BaseModel, Field
    from typing import Any
    from ccam_backend.auth.middleware import verify_hmac
    from ccam_menu.models import MenuItem, MenuConstraints
    from ccam_menu.engine import MenuEngine

    router = APIRouter(tags=["menu"])

    class MenuItemRequest(BaseModel):
        id: str
        name: str
        category: str
        current_price_cents: int = Field(..., gt=0)
        cost_cents: int = Field(..., gt=0)
        avg_daily_orders: float = 1.0

    class OptimizeRequest(BaseModel):
        items: list[MenuItemRequest]

    class PriceChangeResponse(BaseModel):
        item_id: str; item_name: str
        old_price_cents: int; new_price_cents: int
        change_pct: float; reason: str = ""

    class OptimizeResponse(BaseModel):
        changes: list[PriceChangeResponse]
        items_to_remove: list[str]
        projected_profit_cents: int
        current_profit_cents: int
        safety_checks_passed: bool
        solver_status: str
        optimization_score: float
        note: str = "Run menu changes through POST /api/v1/simulate before deploying per D-09"

    @router.post("/menu/optimize", response_model=OptimizeResponse)
    async def optimize_menu(request: Request, body: OptimizeRequest, auth: tuple = Depends(verify_hmac)):
        menu_items = [MenuItem(id=i.id, name=i.name, category=i.category,
            current_price_cents=i.current_price_cents, cost_cents=i.cost_cents,
            margin_cents=i.current_price_cents-i.cost_cents,
            margin_pct=(i.current_price_cents-i.cost_cents)/max(i.cost_cents,1),
            prep_time_seconds=0, avg_daily_orders=i.avg_daily_orders) for i in body.items]
        engine = MenuEngine()
        loop = asyncio.get_event_loop()
        rec = await loop.run_in_executor(None, engine.optimize, menu_items, True)
        return OptimizeResponse(
            changes=[PriceChangeResponse(item_id=c.item_id, item_name=c.item_name,
                old_price_cents=c.old_price_cents, new_price_cents=c.new_price_cents,
                change_pct=c.change_pct, reason=c.reason) for c in rec.changes],
            items_to_remove=rec.items_to_remove,
            projected_profit_cents=rec.projected_profit_cents,
            current_profit_cents=rec.current_profit_cents,
            safety_checks_passed=rec.safety_checks_passed,
            solver_status=rec.solver_status,
            optimization_score=rec.optimization_score,
        )
    ```

        ensemble_quality: float
        note: str = "Validate forecast changes through POST /api/v1/simulate before deploying per D-13"

    class ForecastTrainResponse(BaseModel):
        job_id: str
        status: str = "queued"
        message: str

    @router.post("/forecast/predict", response_model=ForecastPredictResponse)
    async def forecast_predict(request: Request, body: ForecastPredictRequest, auth: tuple = Depends(verify_hmac)):
        gen = SimulatorDataGenerator()
        data = gen.generate_daily_demand(body.item_id, body.item_name, base_volume=body.base_volume, days=120)
        ensemble = ForecastEnsemble()
        loop = asyncio.get_event_loop()
        try:
            await loop.run_in_executor(None, ensemble.train, body.item_id, data, True, True, True)
        except Exception:
            # Fall back to ARIMA-only if full ensemble fails
            await loop.run_in_executor(None, ensemble.train, body.item_id, data, False, False, True)
        fcast = await loop.run_in_executor(None, ensemble.predict, body.horizon_days)
        return ForecastPredictResponse(
            item_id=body.item_id,
            points=[ForecastPointResponse(timestamp=p.timestamp.isoformat(),
                predicted_quantity=p.predicted_quantity, lower_bound=p.lower_bound,
                upper_bound=p.upper_bound) for p in fcast.points],
            weights=fcast.weights,
            ensemble_quality=fcast.ensemble_quality,
        )

    @router.post("/forecast/train", response_model=ForecastTrainResponse)
    async def forecast_train(request: Request, body: dict | None = None, auth: tuple = Depends(verify_hmac)):
        if arq_pool is None:
            raise HTTPException(status_code=503, detail="ARQ pool not initialized")
        item_ids = (body and body.get("item_ids")) or None
        job_id = await arq_pool.enqueue_job("train_forecast_models", item_ids)
        return ForecastTrainResponse(
            job_id=str(job_id) if job_id else "unknown",
            message="Forecast training enqueued. Check job status via GET /api/v1/jobs/{job_id}",
        )

        channel: str; title: str; body: str
        status: str; is_read: bool
        created_at: str; sent_at: str | None

        total: int
        unread_count: int

            db: AsyncSession = Depends(get_db), auth: tuple = Depends(verify_hmac)):
        tenant_id, _ = auth
        channel = NotificationChannel.EMAIL if body.channel == "email" else NotificationChannel.INAPP

            title=body.title, body=body.body, status=results[0].status.value if results else "sent",
            is_read=False, sent_at=datetime.now(timezone.utc),
        )
        db.add(log); await db.commit(); await db.refresh(log)

        return NotificationResponse(id=log.id, tenant_id=log.tenant_id, user_id=log.user_id,
            channel=log.channel, title=log.title, body=log.body, status=log.status,
            is_read=log.is_read, created_at=log.created_at.isoformat() if log.created_at else "",
            sent_at=log.sent_at.isoformat() if log.sent_at else None)

    @router.get("/notifications", response_model=NotificationListResponse)
    async def list_notifications(request: Request, db: AsyncSession = Depends(get_db),

            page_size: int = Query(50, ge=1, le=100)):
        tenant_id, _ = auth
        # Count
        count_q = select(func.count()).select_from(NotificationLog).where(NotificationLog.tenant_id == tenant_id)

        q = q.offset((page-1)*page_size).limit(page_size)
        rows = (await db.execute(q)).scalars().all()
        items = [NotificationResponse(id=r.id, tenant_id=r.tenant_id, user_id=r.user_id,

            sent_at=r.sent_at.isoformat() if r.sent_at else None) for r in rows]
