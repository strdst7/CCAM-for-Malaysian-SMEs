    ```python
    import hmac
    import hashlib
    import base64
    from datetime import datetime, timezone

    def _epoch_ms() -> int:
        return int(datetime.now(timezone.utc).timestamp() * 1000)

    def sign_request(
        secret: bytes,
        method: str,
        path: str,
        body: bytes,
        timestamp: int | None = None,
    ) -> tuple[int, str]:
        if timestamp is None:
            timestamp = _epoch_ms()
        payload = f"{method.upper()}\n{path}\n{timestamp}\n".encode() + body
        signature = hmac.new(secret, payload, hashlib.sha256).digest()
        return timestamp, base64.b64encode(signature).decode()

    def verify_signature(
        secret: bytes,
        method: str,
        path: str,
        body: bytes,
        timestamp: int,
        signature_b64: str,
        max_skew_ms: int = 300_000,
    ) -> bool:
        now_ms = _epoch_ms()
        if abs(now_ms - timestamp) > max_skew_ms:
            return False
        expected_b64 = sign_request(secret, method, path, body, timestamp)[1]
        return hmac.compare_digest(expected_b64, signature_b64)
    ```

    ```python
        if not auth.startswith("HMAC-SHA256 "):
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Missing or malformed Authorization header",
            )

        try:
            credentials = auth[len("HMAC-SHA256 "):]
            signed_key, signature_b64 = credentials.rsplit(":", 1)
        except (ValueError, IndexError):
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Malformed Authorization header",
            )

        timestamp_str = request.headers.get("X-Timestamp", "")
        try:
            timestamp = int(timestamp_str)
        except (ValueError, TypeError):
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Missing or invalid X-Timestamp header",
            )

        key_prefix = signed_key[:8]
        api_key_record = await get_active_key_by_prefix(db, key_prefix)
        if api_key_record is None:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Invalid API key",
            )

        if not verify_key(signed_key, api_key_record.key_hash):
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Invalid API key",
            )

        body_bytes = await request.body()
        secret = signed_key.encode()
        if not verify_signature(secret, request.method, request.url.path, body_bytes, timestamp):
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Invalid signature",
            )

        await db.execute(
            update(ApiKey)
            .where(ApiKey.id == api_key_record.id)
            .values(last_used_at=datetime.now(timezone.utc))
        )

        return api_key_record.tenant_id, signed_key

    # Also export an alias for cleaner imports
    AuthContext = tuple[str, str]  # tenant_id, key_version
    ```

    ```python
    """
    RLS Setup for audit_log immutability.

    This script creates database policies that enforce INSERT-only access
    on the audit_log table. It requires SUPERUSER or appropriate privileges.

    Run: python -m ccam_backend.rls.setup
    """
    import asyncio
    from sqlalchemy import text
    from ccam_backend.database import engine

    RLS_SQL = """
    -- Enable RLS on audit_log table
    ALTER TABLE audit_logs ENABLE ROW LEVEL SECURITY;

    -- Create a role for the application user with restricted privileges
    -- This is advisory - the actual user should be created externally
    DO $$
    BEGIN
        IF NOT EXISTS (SELECT FROM pg_catalog.pg_roles WHERE rolname = 'ccam_app') THEN
            CREATE ROLE ccam_app WITH LOGIN PASSWORD 'ccam_app_password';
        END IF;
    END
    $$;

    -- Grant INSERT only on audit_logs to the application user
    GRANT INSERT ON audit_logs TO ccam_app;
    GRANT USAGE ON SEQUENCE audit_logs_id_seq TO ccam_app;

    -- RLS policy: allow INSERT only
    CREATE POLICY audit_log_insert_only ON audit_logs
        FOR INSERT
        TO ccam_app
        WITH CHECK (true);

    -- Explicitly deny UPDATE and DELETE
    CREATE POLICY audit_log_no_update ON audit_logs
        FOR UPDATE
        TO ccam_app
        WITH CHECK (false);

    CREATE POLICY audit_log_no_delete ON audit_logs
        FOR DELETE
        TO ccam_app
        USING (false);

    -- Grant SELECT for admin users (if needed)
    -- GRANT SELECT ON audit_logs TO ccam_app;
    """

    async def setup_rls() -> None:
        async with engine.connect() as conn:
            for statement in RLS_SQL.split(";"):
                stmt = statement.strip()
                if stmt:
                    await conn.execute(text(stmt + ";"))
            await conn.commit()
        await engine.dispose()

    if __name__ == "__main__":
        asyncio.run(setup_rls())
    ```
