    **`backend/src/ccam_backend/cli/provision.py`** — Tenant provisioning logic:
    ```python
    import asyncio
    import click
    from datetime import datetime, timezone
    from sqlalchemy import select
    from ccam_backend.database import async_session_factory
    from ccam_backend.models.tenant import Tenant
    from ccam_backend.models.user import User
    from ccam_backend.models.api_key import ApiKey
    from ccam_backend.auth.api_keys import generate_api_key
    from ccam_backend.audit.logger import AuditLogger

    @click.command()
    @click.option("--name", "-n", required=True, help="Tenant display name (e.g., 'Kedai Kopi')")
    @click.option("--slug", "-s", help="URL-friendly slug (auto-generated from name if omitted)")
    @click.option("--admin-email", "-e", default="admin@example.com", help="Admin user email")
    @click.option("--admin-password", "-p", default=None, help="Admin password (auto-generated if omitted)")
    @click.option("--api-key-label", "-l", default="default", help="Label for the initial API key")
    def provision_tenant(name, slug, admin_email, admin_password, api_key_label):
        """Provision a new tenant with admin user and API key."""
        asyncio.run(_provision_tenant(name, slug, admin_email, admin_password, api_key_label))

    async def _provision_tenant(name, slug, admin_email, admin_password, api_key_label):
        if slug is None:
            slug = name.lower().replace(" ", "-").replace("_", "-")
        if admin_password is None:
            import secrets, string
            admin_password = ''.join(secrets.choice(string.ascii_letters + string.digits) for _ in range(16))

        async with async_session_factory() as db:
            # Check slug uniqueness
            existing = await db.execute(select(Tenant).where(Tenant.slug == slug))
            if existing.scalar_one_or_none():
                click.echo(f"Error: Tenant with slug '{slug}' already exists", err=True)
                raise click.Abort()

            import bcrypt
            import secrets as sec

            # Create tenant
            tenant = Tenant(name=name, slug=slug, is_active=True)
            db.add(tenant)
            await db.flush()

            # Create admin user
            password_hash = bcrypt.hashpw(admin_password.encode(), bcrypt.gensalt()).decode()
            user = User(
                tenant_id=tenant.id,
                email=admin_email,
                password_hash=password_hash,
                display_name=f"{name} Admin",
                role="admin",
                is_active=True,
            )
            db.add(user)
            await db.flush()

            # Create API key
            raw_key, key_prefix, key_hash, version = generate_api_key()
            api_key = ApiKey(
                tenant_id=tenant.id,
                key_prefix=key_prefix,
                key_hash=key_hash,
                version=version,
                is_active=True,
                label=api_key_label,
            )
            db.add(api_key)
            await db.flush()

            # Audit log
            logger = AuditLogger(db)
            await logger.log(
                tenant_id=tenant.id,
                action="tenant.provision",
                actor_type="system",
                actor_id="cli",
                target_type="tenant",
                target_id=tenant.id,
                after={"name": name, "slug": slug, "admin_email": admin_email},
            )

            await db.commit()

            # Output
            click.echo("─" * 50)
            click.echo("  TENANT PROVISIONED")
            click.echo("─" * 50)
            click.echo(f"  Tenant ID:   {tenant.id}")
            click.echo(f"  Name:        {name}")
            click.echo(f"  Slug:        {slug}")
            click.echo(f"  Admin User:  {admin_email}")
            click.echo(f"  Admin Pass:  {admin_password}")
            click.echo(f"  API Key:     {raw_key}")
            click.echo("─" * 50)
            click.echo("  Store the API key securely — it cannot be retrieved later.")
            click.echo("─" * 50)
    ```

    # Stage 1: Install dependencies with uv
    FROM python:3.14-slim AS builder
    COPY --from=ghcr.io/astral-sh/uv:latest /uv /usr/local/bin/uv

    WORKDIR /app
    COPY pyproject.toml .
    RUN uv sync --frozen --no-dev --no-cache

    # Stage 2: Runtime
    FROM python:3.14-slim

    COPY --from=builder /app/.venv /app/.venv
    COPY src/ src/
    COPY alembic.ini .
    COPY alembic/ alembic/

    ENV PATH="/app/.venv/bin:$PATH"
