    ```python
    """Tests for demand generator — Poisson process arrival generation."""
    import pytest
    from ccam_sim.engines.demand import demand_generator
    from ccam_sim.models import SimState, DemandArrival


    class TestDemandGenerator:
        def test_generates_arrivals(self, empty_state):
            state = demand_generator(empty_state)
            arrivals = [e for e in state.events if isinstance(e, DemandArrival)]
            assert len(arrivals) > 0

        def test_arrivals_have_time_us(self, empty_state):
            state = demand_generator(empty_state)
            for a in [e for e in state.events if isinstance(e, DemandArrival)]:
                assert a.time_us > 0
                assert isinstance(a.time_us, int)

        def test_arrivals_within_window(self, empty_state):
            max_us = int(empty_state.scenario.simulation.total_hours * 3_600_000_000)
            state = demand_generator(empty_state)
            for a in [e for e in state.events if isinstance(e, DemandArrival)]:
                assert a.time_us < max_us

        def test_zero_rate_no_arrivals(self):
            from ccam_sim.models import Scenario, SimulationConfig, DemandBlock
            import random
            s = Scenario(simulation=SimulationConfig(total_hours=8),
                         demand_pattern=[DemandBlock(0, 8, 0, {})],
                         menu=[], stations=[], stock={}, supplier_delays={})
            state = SimState(rng=random.Random(42), scenario=s)
            state = demand_generator(state)
            assert len([e for e in state.events if isinstance(e, DemandArrival)]) == 0

        def test_deterministic(self, simple_scenario):
            import random
            s1 = demand_generator(SimState(rng=random.Random(42), scenario=simple_scenario))
            s2 = demand_generator(SimState(rng=random.Random(42), scenario=simple_scenario))
            t1 = [e.time_us for e in s1.events if isinstance(e, DemandArrival)]
            t2 = [e.time_us for e in s2.events if isinstance(e, DemandArrival)]
            assert t1 == t2

        def test_different_seed(self, simple_scenario):
            import random
            s1 = demand_generator(SimState(rng=random.Random(42), scenario=simple_scenario))
            s2 = demand_generator(SimState(rng=random.Random(99), scenario=simple_scenario))
            t1 = [e.time_us for e in s1.events if isinstance(e, DemandArrival)]
            t2 = [e.time_us for e in s2.events if isinstance(e, DemandArrival)]
    ```

    ```python
    """Tests for kitchen engine."""
    from ccam_sim.engines.demand import demand_generator
    from ccam_sim.engines.order import order_generator
    from ccam_sim.engines.kitchen import kitchen_engine
    from ccam_sim.models import OrderCompleted, StationState


    def _run_ticks(state, n):
        for step in range(n):
            state.step = step
            state.time_us = step * state.scenario.simulation.tick_interval_minutes * 60 * 1_000_000
            state = kitchen_engine(state)
        return state


    class TestKitchenEngine:
        def test_initializes_stations(self, empty_state):
            state = _run_ticks(order_generator(demand_generator(empty_state)), 1)
            assert len(state.station_states) == len(state.scenario.stations)

        def test_orders_complete(self, simple_scenario):
            import random
            from ccam_sim.models import SimState
            state = SimState(rng=random.Random(42), scenario=simple_scenario)
            state = _run_ticks(order_generator(demand_generator(state)), 50)
            assert len([e for e in state.events if isinstance(e, OrderCompleted)]) > 0

        def test_deterministic(self, simple_scenario):
            import random
            def run(seed):
                s = SimState(rng=random.Random(seed), scenario=simple_scenario)
                s = _run_ticks(order_generator(demand_generator(s)), 20)
                return [e.order_id for e in s.events if isinstance(e, OrderCompleted)]
            assert run(42) == run(42)
    ```

    ```python
    from ccam_sim.engines.stock import stock_engine
    ```

    ```python
    """Deterministic replay tests — SIM-08: same YAML + seed = identical output.

    This is the most important test in Phase 1. If this test fails,
    the simulator is NOT deterministic and cannot serve as the system governor.
    """

    import pytest
    import json
    from ccam_sim import run_simulation, load_scenario


    class TestDeterministicReplay:
        """Core determinism tests — SIM-08 requirement."""

        def test_same_seed_same_kpis(self, simple_scenario):
            """Same scenario + seed must produce identical KPI dicts."""
            r1 = run_simulation(simple_scenario, seed=42)
            r2 = run_simulation(simple_scenario, seed=42)
            assert r1.kpis == r2.kpis, (
                f"Determinism violation! KPIs differ with same seed.\n"
                f"Run 1: {r1.kpis}\nRun 2: {r2.kpis}"
            )

        def test_same_seed_same_event_count(self, simple_scenario):
            """Same scenario + seed must produce same number of events."""
            r1 = run_simulation(simple_scenario, seed=42)
            r2 = run_simulation(simple_scenario, seed=42)
            assert len(r1.events) == len(r2.events), (
                f"Event count differs: {len(r1.events)} vs {len(r2.events)}"
            )

        def test_same_seed_same_step_count(self, simple_scenario):
            """Same scenario + seed must execute same number of steps."""
            r1 = run_simulation(simple_scenario, seed=42)
            r2 = run_simulation(simple_scenario, seed=42)
            assert r1.step == r2.step, (
                f"Step count differs: {r1.step} vs {r2.step}"
            )

        def test_different_seed_different_kpis(self, simple_scenario):
            """Different seeds should produce different KPI values (probabilistic)."""
            r1 = run_simulation(simple_scenario, seed=42)
            r2 = run_simulation(simple_scenario, seed=99)
            # It's theoretically possible (extremely unlikely) for different
            # seeds to produce identical KPIs, but this is a sanity check.
            assert r1.kpis != r2.kpis or len(r1.events) != len(r2.events), (
    ```
