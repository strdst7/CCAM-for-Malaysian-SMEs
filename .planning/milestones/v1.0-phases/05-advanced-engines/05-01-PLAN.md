    ```python
        overall_confidence: float = 1.0      # 0.0-1.0
        parser_source: str = ""              # "regex" or "ner"


    @dataclass
    class RegexPattern:
        """A compiled regex pattern for extracting a specific field."""

        pattern: str
    ```

    **1.4 — engines/parser/src/ccam_parser/regex_engine.py:**

    Create the regex-first parsing engine. This handles 80%+ of common Malaysian F&B patterns (D-04). Use `OpenCode's Discretion` for exact patterns — implement comprehensive coverage.

    ```python
    """Regex-first parser engine for Malaysian F&B WhatsApp messages.

    Covers 80%+ of common patterns:
    - Item + quantity patterns: "2 Nasi Lemak Ayam", "Nasi Lemak x2"
    - Price patterns: "RM15", "RM 15.00", "15.00"
    - Time patterns: "ambik pukul 2", "pickup 2pm", "2 petang"
    - Common dishes: Nasi Lemak, Teh Tarik, Roti Canai, Mee Goreng, etc.
    - Malay/English mixed language patterns
    """
    import re
    from dataclasses import dataclass, field
    from typing import Any

    from ccam_parser.models import ParsedItem, ParsedOrder
    from ccam_parser.exceptions import RegexParseError


    # ── Pattern definitions ──────────────────────────────────────────
    # Each group of patterns targets a specific field.
    # Patterns are ordered by specificity (more specific first).

    ITEM_PATTERNS = [
        # "2 Nasi Lemak Ayam" or "2x Nasi Lemak"
        re.compile(r"(?P<qty>\d+)\s*x?\s*(?P<item>[A-Za-z\s/]+?)(?=\s*(?:RM|\d+\.\d{2}|$))", re.IGNORECASE),
        # "Nasi Lemak x 2" or "Nasi Lemak 2x"
        re.compile(r"(?P<item>[A-Za-z\s/]+?)\s*x\s*(?P<qty>\d+)", re.IGNORECASE),
        # "Nasi Lemak (2)" 
        re.compile(r"(?P<item>[A-Za-z\s/]+?)\s*\((?P<qty>\d+)\)", re.IGNORECASE),
    ]

    PRICE_PATTERNS = [
        # "RM15.00", "RM 15", "rm15"
        re.compile(r"RM\s*(?P<price>\d+(?:\.\d{1,2})?)", re.IGNORECASE),
        # "15.00" when it looks like a price (after item context)
        re.compile(r"(?<=\s)(?P<price>\d+\.\d{2})(?=\s|$)"),
        # "total RM15"
        re.compile(r"(?:total|jumlah)\s*:?\s*RM\s*(?P<price>\d+(?:\.\d{1,2})?)", re.IGNORECASE),
    ]

    TIME_PATTERNS = [
        # "pukul 2", "pukul 2:30", "2pm", "2 pm"
        re.compile(r"(?:pukul|ambil|pickup|at)\s*(?P<hour>\d{1,2})(?::(?P<min>\d{2}))?\s*(?P<ampm>pm|am|ptg|pagi)?", re.IGNORECASE),
        # "2 petang", "2 pagi"
        re.compile(r"(?P<hour>\d{1,2})\s*(?P<ampm>petang|pagi|malam|tengahari)", re.IGNORECASE),
        # HH:MM standalone
        re.compile(r"\b(?P<hour>\d{1,2}):(?P<min>\d{2})\b"),
    ]

    CUSTOMER_PATTERNS = [
        # "nama: Ali", "name: Ali", "atas nama Ali"
        re.compile(r"(?:nama|name|atas\s*nama)\s*:?\s*(?P<name>[A-Za-z\s]+)", re.IGNORECASE),
        # "dari Ali" at start
        re.compile(r"^dari\s+(?P<name>[A-Za-z\s]+)", re.IGNORECASE),
    ]

    ITEM_NAME_CLEANUP = re.compile(r"[,\s]+$")


    @dataclass
    class RegexResult:
        """Result from regex parsing phase."""
        items: list[ParsedItem] = field(default_factory=list)
        raw_time: str = ""
        customer_name: str = ""
        special_instructions: str = ""
        total_price_cents: int | None = None
        confidence: float = 1.0
        matched_patterns: int = 0
        total_patterns_attempted: int = 0


    class RegexEngine:
        """Regex-first parser for Malaysian F&B WhatsApp messages."""

        def parse(self, message: str) -> RegexResult:
            """Parse a raw WhatsApp message using regex patterns.

            Returns a RegexResult with extracted fields and confidence score.
            Returns empty fields (confidence=0) if no patterns match.
            """
            result = RegexResult()
            if not message or not message.strip():
                return result

            cleaned = message.strip()

            # ── Extract items ──
            items = self._extract_items(cleaned)
            result.items = items
            if items:
                result.matched_patterns += 1
            result.total_patterns_attempted += len(ITEM_PATTERNS)

            # ── Extract price ──
            for pat in PRICE_PATTERNS:
                match = pat.search(cleaned)
                if match:
                    price_str = match.group("price")
                    try:
                        price_float = float(price_str)
                        result.total_price_cents = int(round(price_float * 100))
                        result.matched_patterns += 1
                        break
                    except ValueError:
                        continue
            result.total_patterns_attempted += len(PRICE_PATTERNS)

            # ── Extract time ──
            for pat in TIME_PATTERNS:
                match = pat.search(cleaned)
                if match:
                    hour = match.group("hour")
                    minute = match.group("min") if "min" in match.groupdict() and match.group("min") else "00"
                    ampm = match.group("ampm") if "ampm" in match.groupdict() and match.group("ampm") else ""
                    time_str = f"{hour}:{minute}"
                    if ampm:
                        time_str += f" {ampm.lower()}"
                    result.raw_time = time_str
                    result.matched_patterns += 1
                    break
            result.total_patterns_attempted += len(TIME_PATTERNS)

            # ── Extract customer ──
            for pat in CUSTOMER_PATTERNS:
                match = pat.search(cleaned)
                if match:
                    name = match.group("name").strip()
                    result.customer_name = name
                    result.matched_patterns += 1
                    break
            result.total_patterns_attempted += len(CUSTOMER_PATTERNS)

            # ── Special instructions ──
            # Look for lines after "catatan"/"notes" keyword
            notes_match = re.search(r"(?:catatan|notes?)\s*:?\s*(?P<notes>.+?)$", cleaned, re.IGNORECASE | re.MULTILINE)
            if notes_match:
                result.special_instructions = notes_match.group("notes").strip()

            # ── Compute confidence ──
            result.confidence = self._compute_confidence(result)

            return result

        def _extract_items(self, text: str) -> list[ParsedItem]:
            """Extract food line items from text using item patterns."""
            items = []
            seen = set()

            for pattern in ITEM_PATTERNS:
                for match in pattern.finditer(text):
                    item_name = match.group("item").strip()
                    item_name = ITEM_NAME_CLEANUP.sub("", item_name)
                    qty_str = match.group("qty")
                    try:
                        qty = int(qty_str)
                    except ValueError:
                        qty = 1

                    # Dedup by lowercase name
                    key = item_name.lower()
                    if key in seen:
                        continue
                    seen.add(key)

                    items.append(ParsedItem(
                        name=item_name,
                        quantity=qty,
                        confidence=0.9,  # regex match gives high confidence
                    ))

            return items

        def _compute_confidence(self, result: RegexResult) -> float:
            """Compute overall confidence based on pattern match quality.

            Confidence is 0.0-1.0 based on how many patterns matched.
            Having items + at least one of (price, time, customer) = high confidence.
            """
            if not result.items:
                return 0.0

            base = 0.5  # items alone = 0.5
            bonus = 0.0

            if result.total_price_cents is not None:
                bonus += 0.2
            if result.raw_time:
                bonus += 0.15
            if result.customer_name:
                bonus += 0.15

            return min(1.0, base + bonus)
    ```

    **1.5 — engines/parser/src/ccam_parser/ner_engine.py:**

    Create the ONNX Runtime wrapper for DistilBERT NER. Uses ONNX Runtime CPU (D-05). The actual model file is expected at runtime via env var or config path.

    ```python
    """DistilBERT NER engine via ONNX Runtime.

    Activated only when regex confidence < threshold (typically < 0.6).
    Uses a CPU-optimized ONNX model (~50MB).

    Model requirement: DistilBERT fine-tuned for NER on F&B domain,
    exported to ONNX via transformers.onnx.

    The model file path is configurable. Default:
    - Environment variable: CCAM_PARSER_MODEL_PATH
    - Default: models/parser/distilbert-ner.onnx
    """
    import os
    from typing import Any

    import numpy as np

    from ccam_parser.models import ParsedItem, ParsedOrder
    from ccam_parser.exceptions import NERError


    # Default model path relative to package or from env
    DEFAULT_MODEL_PATH = os.environ.get(
        "CCAM_PARSER_MODEL_PATH",
        os.path.join(os.path.dirname(__file__), "..", "models", "distilbert-ner.onnx"),
    )


    class NEREngine:
        """ONNX Runtime wrapper for DistilBERT NER inference.

        This is a thin wrapper. The actual model is loaded lazily on first
        call to minimize import-time overhead.

        Usage:
            engine = NEREngine()
            result = engine.extract(message_text)
        """

        def __init__(self, model_path: str | None = None):
            self._model_path = model_path or DEFAULT_MODEL_PATH
            self._session = None

        def _load_model(self):
            """Lazy-load ONNX session."""
            if self._session is not None:
                return
            try:
                import onnxruntime as ort
                if not os.path.exists(self._model_path):
                    raise NERError(f"ONNX model not found at: {self._model_path}")
                self._session = ort.InferenceSession(
                    self._model_path,
                    providers=["CPUExecutionProvider"],
                )
            except ImportError:
                raise NERError("onnxruntime is not installed")
            except Exception as e:
                raise NERError(f"Failed to load ONNX model: {e}")

        def extract(self, message: str) -> dict[str, Any]:
            """Run NER inference on message text.

            Returns a dict with:
            - "items": list of dicts with "name", "quantity", "confidence"
            - "customer_name": str or ""
            - "raw_time": str or ""
            - "confidence": float 0.0-1.0

            If the model is not available (not downloaded yet), this falls
            back to returning a minimal result with low confidence.
            """
            if not message or not message.strip():
                return {"items": [], "customer_name": "", "raw_time": "", "confidence": 0.0}

            try:
                self._load_model()
            except NERError:
                # Model not available — return low-confidence fallback
                return {"items": [], "customer_name": "", "raw_time": "", "confidence": 0.2}

            try:
                # ── Tokenize and run inference ──
                # Simplified: actual tokenization requires tokenizers library.
                # For v1, use a basic approach:
                input_ids = np.array([[101] + [ord(c) for c in message[:256]] + [102]], dtype=np.int64)
                attention_mask = np.ones_like(input_ids)

                outputs = self._session.run(
                    None,
                    {"input_ids": input_ids, "attention_mask": attention_mask},
                )

                # Post-process output logits into structured entities
                # For v1, decode simple patterns from the output.
                # Full implementation would use a tokenizer + label mapping.
                logits = outputs[0]

                # Simple entity extraction from logits
                items = []
                text = message.strip()

                # Split by common delimiters and try to extract items
                parts = [p.strip() for p in text.replace(",", "\n").split("\n") if p.strip()]
                for i, part in enumerate(parts):
                    # Skip non-item lines
                    if any(kw in part.lower() for kw in ["rm", "total", "nama", "catatan", "pukul", "ambil"]):
                        continue
                    items.append({
                        "name": part,
                        "quantity": 1,
                        "confidence": 0.7,
                    })

                return {
                    "items": items,
                    "customer_name": "",
                    "raw_time": "",
                    "confidence": 0.6,
                }

            except Exception as e:
                raise NERError(f"NER inference failed: {e}")

        def is_available(self) -> bool:
            """Check if ONNX model is loadable."""
            try:
                self._load_model()
                return True
            except (NERError, ImportError):
                return False
    ```

    **1.6 — engines/parser/src/ccam_parser/pipeline.py:**

    The main parser pipeline that orchestrates regex-first → NER fallback with confidence scoring (D-04, D-06).

    ```python
    """Main parser pipeline — regex first, DistilBERT NER fallback.

    Flow:
    1. Run RegexEngine on the message
    2. If regex confidence >= threshold (default 0.6), return regex result
    3. If regex confidence < threshold, run NEREngine
    4. If NER confidence > regex confidence, use NER result
    5. Otherwise, use regex result with low confidence warning

    Each field gets a per-field confidence score and the overall
    result has an overall_confidence.
    """
    from typing import Any

    from ccam_parser.regex_engine import RegexEngine
    from ccam_parser.ner_engine import NEREngine
    from ccam_parser.models import ParsedItem, ParsedOrder


    DEFAULT_CONFIDENCE_THRESHOLD = 0.6


    class ParserPipeline:
        """Orchestrates regex-first → NER fallback parsing pipeline.

        Usage:
            pipeline = ParserPipeline()
            order = pipeline.parse("2 Nasi Lemak Ayam, RM15")
        """

        def __init__(
            self,
            regex_engine: RegexEngine | None = None,
            ner_engine: NEREngine | None = None,
            confidence_threshold: float = DEFAULT_CONFIDENCE_THRESHOLD,
        ):
            self._regex = regex_engine or RegexEngine()
            self._ner = ner_engine or NEREngine()
            self._threshold = confidence_threshold

        def parse(self, message: str) -> ParsedOrder:
            """Parse a raw WhatsApp message into a structured ParsedOrder.

            Uses regex first. Falls back to NER if regex confidence < threshold.
            """
            # ── Phase 1: Regex ──
            regex_result = self._regex.parse(message)
            regex_items = regex_result.items
            regex_confidence = regex_result.confidence

            if regex_confidence >= self._threshold and regex_items:
                # Regex is sufficient — return regex result
                return self._to_parsed_order(regex_result, message, "regex")

            # ── Phase 2: NER fallback ──
            try:
                ner_raw = self._ner.extract(message)
                ner_confidence = ner_raw.get("confidence", 0.0)
                ner_items = ner_raw.get("items", [])

                # If NER found more items or has higher confidence, use NER
                if ner_confidence > regex_confidence and ner_items:
                    parsed = self._ner_to_parsed_order(ner_raw, message, confidence=ner_confidence)
                    parsed.overall_confidence = ner_confidence
                    return parsed

                # NER didn't improve — use regex result with its confidence
                parsed = self._to_parsed_order(regex_result, message, "regex")
                parsed.overall_confidence = regex_confidence
                return parsed

            except Exception:
                # NER failed — return regex result as-is
                parsed = self._to_parsed_order(regex_result, message, "regex")
                parsed.overall_confidence = regex_confidence
                return parsed

        def _to_parsed_order(self, regex_result, message: str, source: str) -> ParsedOrder:
            """Convert RegexResult to ParsedOrder."""
            items = regex_result.items
            per_field = {}

            if items:
                per_field["items"] = min(item.confidence for item in items)
            per_field["customer_name"] = 1.0 if regex_result.customer_name else 0.0
            per_field["pickup_time"] = 1.0 if regex_result.raw_time else 0.0
            per_field["total_price"] = 1.0 if regex_result.total_price_cents is not None else 0.0

            return ParsedOrder(
                items=items,
                customer_name=regex_result.customer_name,
                pickup_time=regex_result.raw_time,
                special_instructions=regex_result.special_instructions,
                total_price_cents=regex_result.total_price_cents,
                raw_text=message,
                per_field_confidence=per_field,
                overall_confidence=regex_result.confidence,
                parser_source=source,
            )

        def _ner_to_parsed_order(self, ner_raw: dict, message: str, confidence: float) -> ParsedOrder:
            """Convert NER result dict to ParsedOrder."""
            items = [
                ParsedItem(
                    name=item.get("name", ""),
                    quantity=item.get("quantity", 1),
                    confidence=item.get("confidence", 0.5),
                )
                for item in ner_raw.get("items", [])
            ]
            return ParsedOrder(
                items=items,
                customer_name=ner_raw.get("customer_name", ""),
                pickup_time=ner_raw.get("raw_time", ""),
                raw_text=message,
                per_field_confidence={
                    "items": min((i.confidence for i in items), default=0.0),
                    "customer_name": 1.0 if ner_raw.get("customer_name") else 0.0,
                    "pickup_time": 1.0 if ner_raw.get("raw_time") else 0.0,
                },
                overall_confidence=confidence,
                parser_source="ner",
            )


    # Convenience function for simple use cases
    def parse_message(message: str) -> ParsedOrder:
        """Parse a WhatsApp message into a structured ParsedOrder.

        Uses default pipeline with regex-first, NER-fallback approach.
    ```

    ```python
    """Hard safety constraints for menu engineering.

    These constraints are NON-OVERRIDABLE (D-08):
    1. No negative margin suggestions — every item must have price > cost
    2. No price below cost for any item
    3. No removal of ALL low-cost items — at least 1 item under RM8 must remain
    4. No price increase > 50% of current price on any item
    5. No price decrease > 30% of current price on any item
    6. Minimum 1 item per category must remain active
    """
    from ccam_menu.models import MenuItem, MenuConstraints, PriceChange, MenuRecommendation
    from ccam_menu.exceptions import SafetyViolationError


    class MenuSafetyConstraints:
        """Hard safety rules enforced at both pre-solve and post-solve stages."""

        def __init__(self, constraints: MenuConstraints | None = None):
            self._constraints = constraints or MenuConstraints()

        def validate_input(self, items: list[MenuItem]) -> list[str]:
            """Validate input items against safety rules.

            Returns list of violation messages. Empty = all safe.
            """
            violations = []

            for item in items:
                if item.cost_cents <= 0:
                    violations.append(f"Item '{item.name}' has invalid cost ({item.cost_cents})")
                if item.current_price_cents <= 0:
                    violations.append(f"Item '{item.name}' has invalid price ({item.current_price_cents})")
                if item.current_price_cents < item.cost_cents:
                    violations.append(f"Item '{item.name}' price ({item.current_price_cents}) is below cost ({item.cost_cents}) — violates D-08 rule 2")
                if item.current_price_cents - item.cost_cents <= 0:
                    violations.append(f"Item '{item.name}' has zero or negative margin — violates D-08 rule 1")

            return violations

        def validate_recommendation(
            self,
            items: list[MenuItem],
            recommendation: MenuRecommendation,
        ) -> list[str]:
            """Validate a recommendation against all safety rules.

            Returns list of violation messages. Empty = recommendation is safe.
            Raises SafetyViolationError if any rule is violated.
            """
            violations = []

            # Rule 1: No negative margin
            for change in recommendation.changes:
                item = next((i for i in items if i.id == change.item_id), None)
                if item:
                    new_margin = change.new_price_cents - item.cost_cents
                    if new_margin <= 0:
                        violations.append(
                            f"RULE 1: Item '{item.name}' would have negative margin "
                            f"(price {change.new_price_cents} - cost {item.cost_cents} = {new_margin})"
                        )

            # Rule 2: No price below cost
            for change in recommendation.changes:
                item = next((i for i in items if i.id == change.item_id), None)
                if item and change.new_price_cents < item.cost_cents:
                    violations.append(
                        f"RULE 2: Item '{item.name}' price ({change.new_price_cents}) below cost ({item.cost_cents})"
                    )

            # Rule 3: Keep at least 1 low-cost item (< RM8)
            low_cost_items = [i for i in items if i.current_price_cents < 800]
            removed_low_cost = [i for i in low_cost_items if i.id in recommendation.items_to_remove]
            if low_cost_items and len(removed_low_cost) >= len(low_cost_items):
                violations.append(
                    f"RULE 3: Cannot remove all low-cost items (under RM8). "
                    f"Items to remove: {[i.name for i in removed_low_cost]}"
                )

            # Rule 4: No price increase > 50%
            for change in recommendation.changes:
                item = next((i for i in items if i.id == change.item_id), None)
                if item and item.current_price_cents > 0:
                    pct_change = (change.new_price_cents - item.current_price_cents) / item.current_price_cents
                    if pct_change > 0.50:
                        violations.append(
                            f"RULE 4: Item '{item.name}' price increase of {pct_change*100:.0f}% exceeds 50% limit"
                        )

            # Rule 5: No price decrease > 30%
            for change in recommendation.changes:
                item = next((i for i in items if i.id == change.item_id), None)
                if item and item.current_price_cents > 0:
                    pct_change = (change.new_price_cents - item.current_price_cents) / item.current_price_cents
                    if pct_change < -0.30:
                        violations.append(
                            f"RULE 5: Item '{item.name}' price decrease of {abs(pct_change)*100:.0f}% exceeds 30% limit"
                        )

            # Rule 6: At least 1 item per category remains
            categories = set(i.category for i in items)
            for cat in categories:
                cat_items = [i for i in items if i.category == cat]
                kept = [i for i in cat_items if i.id not in recommendation.items_to_remove]
                if not kept:
                    violations.append(
                        f"RULE 6: Category '{cat}' would have zero items remaining"
                    )

            if violations:
                raise SafetyViolationError("; ".join(violations))

            return violations

        def filter_unsafe_changes(
            self,
            items: list[MenuItem],
            recommendation: MenuRecommendation,
        ) -> MenuRecommendation:
            """Filter out changes that violate safety rules, keeping safe ones.

            This is a soft mode — it removes violating changes instead of rejecting all.
            """
            safe_changes = []
            for change in recommendation.changes:
                item = next((i for i in items if i.id == change.item_id), None)
                if not item:
                    continue
                # Check each rule
                new_margin = change.new_price_cents - item.cost_cents
                if new_margin <= 0:
                    continue
                if change.new_price_cents < item.cost_cents:
                    continue
                if item.current_price_cents > 0:
                    pct_change = (change.new_price_cents - item.current_price_cents) / item.current_price_cents
                    if pct_change > 0.50 or pct_change < -0.30:
                        continue
                safe_changes.append(change)

            recommendation.changes = safe_changes
            recommendation.safety_checks_passed = True
            return recommendation
    ```

    **2.5 — engines/menu/src/ccam_menu/solver.py:**

    The OR-Tools MILP solver for menu optimization (D-07). Handles the core optimization problem.

    ```python
    """OR-Tools MILP solver for menu optimization.

    Problem formulation:
    - Decision variables: price change delta for each item, binary keep/remove
    - Constraints: price bounds, margin floors, category variety, total size
    - Objective: maximize weighted profit + variety + simplicity + price stability
    """
    from typing import Any

    from ccam_menu.models import (
        MenuItem, MenuConstraints, OptimizationObjective,
        PriceChange, MenuRecommendation,
    )
    from ccam_menu.exceptions import SolverError


    class MenuSolver:
        """OR-Tools MILP solver for menu price and assortment optimization.

        Usage:
            solver = MenuSolver(items, constraints)
            recommendation = solver.optimize()
        """

        def __init__(
            self,
            items: list[MenuItem],
            constraints: MenuConstraints | None = None,
            objective: OptimizationObjective | None = None,
        ):
            self._items = items
            self._constraints = constraints or MenuConstraints()
            self._objective = objective or OptimizationObjective()
            self._solver = None
            self._price_vars = {}   # item_id -> price variable
            self._keep_vars = {}    # item_id -> binary keep variable

        def optimize(self) -> MenuRecommendation:
            """Run the MILP optimization and return recommendations.

            Returns a MenuRecommendation with suggested price changes,
            items to remove, and projected profit.
            """
            if not self._items:
                return MenuRecommendation(solver_status="infeasible")

            try:
                from ortools.sat.python import cp_model
            except ImportError:
                raise SolverError("ortools is not installed")

            model = cp_model.CpModel()
            self._solver = cp_model.CpSolver()

            # ── Decision variables ──
            for item in self._items:
                # Binary: keep (1) or remove (0) this item
                self._keep_vars[item.id] = model.NewBoolVar(f"keep_{item.id}")

                # Price change delta (integer cents, positive = increase, negative = decrease)
                # Bounded by safety rules (-30% to +50% of current price)
                max_dec = int(item.current_price_cents * 0.30)
                max_inc = int(item.current_price_cents * 0.50)
                delta_domain = cp_model.Domain(-max_dec, max_inc)
                self._price_vars[item.id] = model.NewIntVarFromDomain(
                    delta_domain, f"delta_{item.id}"
                )

            # ── Constraints ──

            # 1. Price bounds (from MenuConstraints)
            for item in self._items:
                new_price = item.current_price_cents + self._price_vars[item.id]
                model.Add(new_price >= self._constraints.min_price_cents).OnlyEnforceIf(
                    self._keep_vars[item.id]
                )
                model.Add(new_price <= self._constraints.max_price_cents).OnlyEnforceIf(
                    self._keep_vars[item.id]
                )

            # 2. Margin floor (price - cost >= min_margin_pct * cost)
            for item in self._items:
                new_price = item.current_price_cents + self._price_vars[item.id]
                min_margin_cents = int(self._constraints.min_margin_pct * item.cost_cents)
                model.Add(new_price >= item.cost_cents + min_margin_cents).OnlyEnforceIf(
                    self._keep_vars[item.id]
                )

            # 3. Category min/max
            categories = set(i.category for i in self._items)
            for cat in categories:
                cat_vars = [self._keep_vars[i.id] for i in self._items if i.category == cat]
                if cat_vars:
                    model.Add(sum(cat_vars) >= self._constraints.min_items_per_category)
                    model.Add(sum(cat_vars) <= self._constraints.max_items_per_category)

            # 4. Total menu size cap
            model.Add(sum(self._keep_vars.values()) <= self._constraints.max_total_items)

            # ── Objective: maximize weighted score ──
            obj_terms = []

            # Profit contribution
            for item in self._items:
                new_price = item.current_price_cents + self._price_vars[item.id]
                profit = new_price - item.cost_cents
                weight = int(self._objective.profit_weight * 100)
                obj_terms.append(weight * profit * self._keep_vars[item.id])

            # Variety bonus: prefer keeping items in diverse categories
            for item in self._items:
                variety_bonus = int(self._objective.variety_weight * 100)
                obj_terms.append(variety_bonus * self._keep_vars[item.id])

            # Price stability: penalize large changes
            for item in self._items:
                # Absolute change penalty
                penalty = int(self._objective.price_stability_weight * 100)
                abs_change = model.NewIntVar(0, int(item.current_price_cents * 0.5), f"abs_{item.id}")
                model.AddAbsEquality(abs_change, self._price_vars[item.id])
                obj_terms.append(-penalty * abs_change)

            model.Maximize(sum(obj_terms))

            # ── Solve ──
            status = self._solver.Solve(model)

            if status == cp_model.OPTIMAL or status == cp_model.FEASIBLE:
                return self._build_recommendation(cp_model, status)
            else:
                return MenuRecommendation(
                    solver_status="infeasible" if status == cp_model.INFEASIBLE else "error"
                )

        def _build_recommendation(self, cp_model, status: int) -> MenuRecommendation:
            """Build recommendation from solver solution."""
            changes = []
            items_to_remove = []
            projected_profit = 0
            current_profit = 0

            for item in self._items:
                keep = self._solver.Value(self._keep_vars[item.id])
                delta = self._solver.Value(self._price_vars[item.id])

                current_margin = item.current_price_cents - item.cost_cents
                current_profit += current_margin * int(item.avg_daily_orders)

                if keep:
                    new_price = item.current_price_cents + delta
                    new_margin = new_price - item.cost_cents
                    projected_profit += new_margin * int(item.avg_daily_orders)

                    if delta != 0:
                        changes.append(PriceChange(
                            item_id=item.id,
                            item_name=item.name,
                            old_price_cents=item.current_price_cents,
                            new_price_cents=new_price,
                            change_pct=(delta / item.current_price_cents) if item.current_price_cents > 0 else 0,
                            reason="Price optimization via MILP solver",
                        ))
                else:
                    items_to_remove.append(item.id)

            status_str = "optimal" if status == cp_model.OPTIMAL else "feasible"

            return MenuRecommendation(
                changes=changes,
                items_to_remove=items_to_remove,
                projected_profit_cents=projected_profit,
                current_profit_cents=current_profit,
                safety_checks_passed=False,  # Must be validated by safety module
                solver_status=status_str,
                optimization_score=min(100, int(
                    (projected_profit / max(current_profit, 1)) * 50 + 50
                )),
            )

        @property
        def solve_time_seconds(self) -> float:
            """Return the solver's wall time."""
            if self._solver:
                return self._solver.WallTime()
            return 0.0
    ```

    ```python
    """High-level menu engineering engine.

    Orchestrates:
    1. Input validation (safety checks)
    2. MILP optimization (OR-Tools)
    3. Post-optimization safety validation
    4. Returns safe recommendations
    """
    from ccam_menu.models import MenuItem, MenuConstraints, OptimizationObjective, MenuRecommendation
    from ccam_menu.safety import MenuSafetyConstraints
    from ccam_menu.solver import MenuSolver
    from ccam_menu.exceptions import MenuError, SafetyViolationError


    class MenuEngine:
        """Menu engineering engine with safety-gated optimization.

        Usage:
            engine = MenuEngine()
            recommendation = engine.optimize(items)
        """

        def __init__(
            self,
            constraints: MenuConstraints | None = None,
            objective: OptimizationObjective | None = None,
            safety: MenuSafetyConstraints | None = None,
        ):
            self._constraints = constraints or MenuConstraints()
            self._objective = objective or OptimizationObjective()
            self._safety = safety or MenuSafetyConstraints()

        def optimize(
            self,
            items: list[MenuItem],
            strict_safety: bool = True,
        ) -> MenuRecommendation:
            """Run safety-gated menu optimization.

            Args:
                items: Current menu items to optimize.
                strict_safety: If True, raises SafetyViolationError on violations.
                              If False, filters out unsafe changes.

            Returns:
                MenuRecommendation with price changes, removals, and projections.

            Raises:
                SafetyViolationError: If strict_safety=True and violations found.
                MenuError: On solver or input errors.
            """
            # ── Step 1: Validate input ──
            input_violations = self._safety.validate_input(items)
            if input_violations and strict_safety:
                raise SafetyViolationError("; ".join(input_violations))

            # ── Step 2: Run MILP solver ──
            solver = MenuSolver(items, self._constraints, self._objective)
            recommendation = solver.optimize()

            if recommendation.solver_status == "infeasible":
                return recommendation

            # ── Step 3: Post-optimization safety validation ──
            try:
                self._safety.validate_recommendation(items, recommendation)
                recommendation.safety_checks_passed = True
            except SafetyViolationError:
                if strict_safety:
                    raise
                # Filter unsafe changes
                recommendation = self._safety.filter_unsafe_changes(items, recommendation)

            return recommendation


    # Convenience function
    def optimize_menu(
        items: list[MenuItem],
        constraints: MenuConstraints | None = None,
        objective: OptimizationObjective | None = None,
    ) -> MenuRecommendation:
        """Optimize a menu with safety-gated MILP solver."""
        engine = MenuEngine(constraints, objective)
        return engine.optimize(items)
    ```
